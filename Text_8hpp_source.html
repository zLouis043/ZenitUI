261 void wrapText(float availW, float fs, float sp, FontHandle f)
262 {
263 lines.clear();
264 auto r = UIContext::get().renderer;
265 if (!r)
266 return;
267
268 auto measure = [&](std::string_view s) -> Vec2
269 {
270 return r->measureText(f, s, fs, sp);
271 };
272
273 // Append `word` to `cur`, or flush and start a new line, or break the
274 // word into chunks when it is wider than `availW`. No recursion.
275 auto emitWord = [&](std::string_view word, std::string &cur)
276 {
277 if (word.empty())
278 return;
279
280 // Fast path: try appending " word" to the current line.
281 if (!cur.empty())
282 {
283 std::string candidate = cur + " " + std::string(word);
284 if (measure(candidate).x <= availW)
285 {
286 cur = std::move(candidate);
287 return;
288 }
289 // Does not fit: flush and fall through with cur cleared.
290 Vec2 s = measure(cur);
291 lines.push_back({cur, s});
292 cur.clear();
293 }
294
295 // Fast path: word fits on an empty line.
296 if (measure(word).x <= availW)
297 {
298 cur.assign(word);
299 return;
300 }
301
302 // Slow path: break the word into chunks that fit (grapheme-cluster safe).
303 std::string_view remaining = word;
304 while (!remaining.empty())
305 {
306 auto bounds = Utf8::clusterBoundaries(remaining);
307 // Longest cluster prefix that still fits in availW (binary search).
308 size_t lo = 1;
309 size_t hi = bounds.size() - 1;
310 size_t bestIdx = 0;
311 while (lo <= hi)
312 {
313 size_t mid = (lo + hi) / 2;
314 if (measure(remaining.substr(0, bounds[mid])).x <= availW)
315 {
316 bestIdx = mid;
317 lo = mid + 1;
318 }
319 else
320 {
321 hi = mid - 1;
322 }
323 }
324 if (bestIdx == 0)
325 bestIdx = 1; // always advance by at least 1 grapheme cluster
326
327 size_t bestBytes = bounds[bestIdx];
328 if (bestBytes >= remaining.size())
329 {
330 cur.assign(remaining);
331 return;
332 }
333
334 std::string chunk(remaining.substr(0, bestBytes));
335 Vec2 s = measure(chunk);
336 lines.push_back({chunk, s});
337 remaining.remove_prefix(bestBytes);
338 }
339 };
340
341 // Split the text on explicit '\n' into logical lines, then wrap each one.
342 size_t lineStart = 0;
343 while (lineStart <= text.size())
344 {
345 size_t lineEnd = text.find('\n', lineStart);
346 std::string_view logicalLine = (lineEnd == std::string::npos)
347 ? std::string_view(text).substr(lineStart)
348 : std::string_view(text).substr(lineStart, lineEnd - lineStart);
349
350 std::string cur;
351 size_t i = 0;
352 while (i < logicalLine.size())
353 {
354 while (i < logicalLine.size() && logicalLine[i] == ' ')
355 ++i;
356 if (i >= logicalLine.size())
357 break;
358
359 size_t wordStart = i;
360 while (i < logicalLine.size() && logicalLine[i] != ' ')
361 ++i;
362 emitWord(logicalLine.substr(wordStart, i - wordStart), cur);
363 }
364
365 if (!cur.empty())
366 {
367 Vec2 s = measure(cur);
368 lines.push_back({cur, s});
369 }
370 else if (logicalLine.empty())
371 {
372 // Preserve empty lines produced by consecutive '\n'.
373 lines.push_back({"", measure("")});
374 }
375
376 if (lineEnd == std::string::npos)
377 break;
378 lineStart = lineEnd + 1;
379 }
380
381 if (lines.empty())
382 {
383 Vec2 sz = measure(text);
384 lines.push_back({text, sz});
385 }
386 }

395 std::string buildEllipsisText(std::string_view src, float availW,
396 FontHandle f, float fs, float sp)
397 {
398 auto r = UIContext::get().renderer;
399 if (!r)
400 return std::string(src);
401
402 // If even the bare ellipsis does not fit, shrink it.
403 Vec2 ellipsisOnly = r->measureText(f, "...", fs, sp);
404 if (ellipsisOnly.x > availW)
405 {
406 if (r->measureText(f, "..", fs, sp).x <= availW)
407 return "..";
408 if (r->measureText(f, ".", fs, sp).x <= availW)
409 return ".";
410 return "";
411 }
412
413 // Longest cluster prefix such that (prefix + "...") fits in availW.
414 auto bounds = Utf8::clusterBoundaries(src);
415 size_t lo = 0, hi = bounds.size() - 1, bestIdx = 0;
416 while (lo <= hi)
417 {
418 size_t mid = (lo + hi) / 2;
419 std::string candidate = std::string(src.substr(0, bounds[mid])) + "...";
420 if (r->measureText(f, candidate, fs, sp).x <= availW)
421 {
422 bestIdx = mid;
423 lo = mid + 1;
424 }
425 else
426 {
427 hi = mid - 1;
428 }
429 }
430 return std::string(src.substr(0, bounds[bestIdx])) + "...";
431 }

440 Vec2 measureSingleLine(FontHandle f)
441 {
442 float fs = style_.currentStyle.fontSize.resolveV(Metrics::viewport.x, Metrics::viewport.y);
443 float sp = style_.currentStyle.letterSpacing.resolveH(Metrics::viewport.x, Metrics::viewport.y);
444
445 if (text_dirty || f.id != cachedFontId ||
446 fs != cachedFs || sp != cachedSp || cachedSize.y == 0.0f ||
447 cachedWrapW >= 0.0f)
448 {
449 auto r = UIContext::get().renderer;
450 lines.clear();
451
452 // Split on explicit '\n' even without wrapping. This makes
453 // '\n' produce a hard line break in every mode (wrap:true or
454 // wrap:false), matching CSS.
455 size_t lineStart = 0;
456 while (lineStart <= text.size())
457 {
458 size_t lineEnd = text.find('\n', lineStart);
459 std::string_view part = (lineEnd == std::string::npos)
460 ? std::string_view(text).substr(lineStart)
461 : std::string_view(text).substr(lineStart, lineEnd - lineStart);
462
463 Vec2 sz = r->measureText(f, part, fs, sp);
464 lines.push_back({std::string(part), sz});
465
466 if (lineEnd == std::string::npos)
467 break;
468 lineStart = lineEnd + 1;
469 }
470
471 float maxW = 0.0f, totalH = 0.0f;
472 for (auto &ln : lines)
473 {
474 maxW = std::max(maxW, ln.size.x);
475 totalH += ln.size.y;
476 }
477 cachedSize = {maxW, totalH};
478 cachedFs = fs;
479 cachedSp = sp;
480 cachedFontId = f.id;
481 cachedWrapW = -1.0f;
482 text_dirty = false;
483 }
484 return cachedSize;
485 }

504 cachedFs = fs;
505 cachedSp = sp;
506 cachedWrapW = availW;
507 cachedFontId = f.id;
508 text_dirty = false;
509 }
510
511 float maxW = 0.0f, totalH = 0.0f;
512 for (auto &ln : lines)
513 {
514 maxW = std::max(maxW, ln.size.x);
515 totalH += ln.size.y;
516 }
