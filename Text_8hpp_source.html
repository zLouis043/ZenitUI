123 float fs = style.fontSize.resolveV(Metrics::viewport.x, Metrics::viewport.y);
124 float sp = style.letterSpacing.resolveH(Metrics::viewport.x, Metrics::viewport.y);
125
126 float bwl = style.borderWidth.left.resolveH(rect.width, rect.height);
127 float bwr = style.borderWidth.right.resolveH(rect.width, rect.height);
128 float bwt = style.borderWidth.top.resolveV(rect.width, rect.height);
129 float bwb = style.borderWidth.bottom.resolveV(rect.width, rect.height);
130
131 float pl = style.padding.left.resolveH(rect.width, rect.height) + bwl;
132 float pr = style.padding.right.resolveH(rect.width, rect.height) + bwr;
133 float pt = style.padding.top.resolveV(rect.width, rect.height) + bwt;
134 float pb = style.padding.bottom.resolveV(rect.width, rect.height) + bwb;
135
136 float availW = std::max(0.0f, rect.width - pl - pr);
137 float availH = std::max(0.0f, rect.height - pt - pb);

139 // Ellipsis: only meaningful for single-line text (wrap disabled or
140 // white-space: nowrap) that overflows the available width.
141 const bool isSingleLine = lines.size() == 1;
142 const bool applyEllipsis =
143 isSingleLine &&
144 style.textOverflow == TextOverflow::Ellipsis &&
145 lines[0].size.x > availW;
146
147 std::string truncatedText;
148 Vec2 truncatedSize = lines[0].size;
149 if (applyEllipsis)
150 {
151 truncatedText = buildEllipsisText(lines[0].text, availW, f, fs, sp);
152 truncatedSize = UIContext::get().renderer->measureText(

154 }
155

157 if (applyEllipsis)
158 totalTextH = truncatedSize.y;
159 else
160 for (auto &ln : lines)
161 totalTextH += ln.size.y;

165
166 if (style.verticalAlign == Align::Center)

168 else if (style.verticalAlign == Align::End)
169 pos.y += availH - totalTextH;
170
171 auto r = UIContext::get().renderer;
172 r->pushClip(rect);
173
174 if (applyEllipsis)
175 {
176 float x = pos.x;
177 if (ha == Align::Center)
178 x += (availW - truncatedSize.x) * 0.5f;
179 else if (ha == Align::End)
180 x += availW - truncatedSize.x;
181 r->drawText(f, truncatedText, {x, pos.y}, fs, sp,
182 style.color.withAlpha(op));
183 }
184 else
185 {
186 float curY = pos.y;
187 for (auto &ln : lines)
188 {
189 float x = pos.x;
190 if (ha == Align::Center)
191 x += (availW - ln.size.x) * 0.5f;
192 else if (ha == Align::End)
193 x += availW - ln.size.x;
194 r->drawText(f, ln.text, {x, curY}, fs, sp,
195 style.color.withAlpha(op));

265 void wrapText(float availW, float fs, float sp, FontHandle f)
266 {
267 lines.clear();
268 auto r = UIContext::get().renderer;
269 if (!r)
270 return;
271
272 auto measure = [&](std::string_view s) -> Vec2
273 {
274 return r->measureText(f, s, fs, sp);
275 };
276
277 // Append `word` to `cur`, or flush and start a new line, or break the
278 // word into chunks when it is wider than `availW`. No recursion.
279 auto emitWord = [&](std::string_view word, std::string &cur)
280 {
281 if (word.empty())
282 return;
283
284 // Fast path: try appending " word" to the current line.
285 if (!cur.empty())
286 {
287 std::string candidate = cur + " " + std::string(word);
288 if (measure(candidate).x <= availW)
289 {
290 cur = std::move(candidate);
291 return;
292 }
293 // Does not fit: flush and fall through with cur cleared.
294 Vec2 s = measure(cur);
295 lines.push_back({cur, s});
296 cur.clear();
297 }
298
299 // Fast path: word fits on an empty line.
300 if (measure(word).x <= availW)
301 {
302 cur.assign(word);
303 return;
304 }
305
306 // Slow path: break the word into chunks that fit (grapheme-cluster safe).
307 std::string_view remaining = word;
308 while (!remaining.empty())
309 {
310 auto bounds = Utf8::clusterBoundaries(remaining);
311 // Longest cluster prefix that still fits in availW (binary search).
312 size_t lo = 1;
313 size_t hi = bounds.size() - 1;
314 size_t bestIdx = 0;
315 while (lo <= hi)
316 {
317 size_t mid = (lo + hi) / 2;
318 if (measure(remaining.substr(0, bounds[mid])).x <= availW)
319 {
320 bestIdx = mid;
321 lo = mid + 1;
322 }
323 else
324 {
325 hi = mid - 1;
326 }
327 }
328 if (bestIdx == 0)
329 bestIdx = 1; // always advance by at least 1 grapheme cluster
330
331 size_t bestBytes = bounds[bestIdx];
332 if (bestBytes >= remaining.size())
333 {
334 cur.assign(remaining);
335 return;
336 }
337
338 std::string chunk(remaining.substr(0, bestBytes));
339 Vec2 s = measure(chunk);
340 lines.push_back({chunk, s});
341 remaining.remove_prefix(bestBytes);
342 }
343 };
344
345 // Split the text on explicit '\n' into logical lines, then wrap each one.
346 size_t lineStart = 0;
347 while (lineStart <= text.size())
348 {
349 size_t lineEnd = text.find('\n', lineStart);
350 std::string_view logicalLine = (lineEnd == std::string::npos)
351 ? std::string_view(text).substr(lineStart)
352 : std::string_view(text).substr(lineStart, lineEnd - lineStart);
353
354 std::string cur;
355 size_t i = 0;
356 while (i < logicalLine.size())
357 {
358 while (i < logicalLine.size() && logicalLine[i] == ' ')
359 ++i;
360 if (i >= logicalLine.size())
361 break;
362
363 size_t wordStart = i;
364 while (i < logicalLine.size() && logicalLine[i] != ' ')
365 ++i;
366 emitWord(logicalLine.substr(wordStart, i - wordStart), cur);
367 }
368
369 if (!cur.empty())
370 {
371 Vec2 s = measure(cur);
372 lines.push_back({cur, s});
373 }
374 else if (logicalLine.empty())
375 {
376 // Preserve empty lines produced by consecutive '\n'.
377 lines.push_back({"", measure("")});
378 }
379
380 if (lineEnd == std::string::npos)
381 break;
382 lineStart = lineEnd + 1;
383 }
384
385 if (lines.empty())
386 {
387 Vec2 sz = measure(text);
388 lines.push_back({text, sz});
389 }
390 }

399 std::string buildEllipsisText(std::string_view src, float availW,
400 FontHandle f, float fs, float sp)
401 {
402 auto r = UIContext::get().renderer;
403 if (!r)
404 return std::string(src);
405
406 // If even the bare ellipsis does not fit, shrink it.
407 Vec2 ellipsisOnly = r->measureText(f, "...", fs, sp);
408 if (ellipsisOnly.x > availW)
409 {
410 if (r->measureText(f, "..", fs, sp).x <= availW)
411 return "..";
412 if (r->measureText(f, ".", fs, sp).x <= availW)
413 return ".";
414 return "";
415 }
416
417 // Longest cluster prefix such that (prefix + "...") fits in availW.
418 auto bounds = Utf8::clusterBoundaries(src);
419 size_t lo = 0, hi = bounds.size() - 1, bestIdx = 0;
420 while (lo <= hi)
421 {
422 size_t mid = (lo + hi) / 2;
423 std::string candidate = std::string(src.substr(0, bounds[mid])) + "...";
424 if (r->measureText(f, candidate, fs, sp).x <= availW)
425 {
426 bestIdx = mid;
427 lo = mid + 1;
428 }
429 else
430 {
431 hi = mid - 1;
432 }
433 }
434 return std::string(src.substr(0, bounds[bestIdx])) + "...";
435 }

444 Vec2 measureSingleLine(FontHandle f)
445 {
446 float fs = style_.currentStyle.fontSize.resolveV(Metrics::viewport.x, Metrics::viewport.y);
447 float sp = style_.currentStyle.letterSpacing.resolveH(Metrics::viewport.x, Metrics::viewport.y);
448
449 if (text_dirty || f.id != cachedFontId ||
450 fs != cachedFs || sp != cachedSp || cachedSize.y == 0.0f ||
451 cachedWrapW >= 0.0f)
452 {
453 auto r = UIContext::get().renderer;
454 lines.clear();
455
456 // Split on explicit '\n' even without wrapping. This makes
457 // '\n' produce a hard line break in every mode (wrap:true or
458 // wrap:false), matching CSS.
459 size_t lineStart = 0;
460 while (lineStart <= text.size())
461 {
462 size_t lineEnd = text.find('\n', lineStart);
463 std::string_view part = (lineEnd == std::string::npos)
464 ? std::string_view(text).substr(lineStart)
465 : std::string_view(text).substr(lineStart, lineEnd - lineStart);
466
467 Vec2 sz = r->measureText(f, part, fs, sp);
468 lines.push_back({std::string(part), sz});
469
470 if (lineEnd == std::string::npos)
471 break;
472 lineStart = lineEnd + 1;
473 }
474
475 float maxW = 0.0f, totalH = 0.0f;
476 for (auto &ln : lines)
477 {
478 maxW = std::max(maxW, ln.size.x);
479 totalH += ln.size.y;
480 }
481 cachedSize = {maxW, totalH};
482 cachedFs = fs;
483 cachedSp = sp;
484 cachedFontId = f.id;
485 cachedWrapW = -1.0f;
486 text_dirty = false;
487 }
488 return cachedSize;
489 }

498 Vec2 measureWrapped(float availW, FontHandle f)
499 {
500 float fs = style_.currentStyle.fontSize.resolveV(Metrics::viewport.x, Metrics::viewport.y);
501 float sp = style_.currentStyle.letterSpacing.resolveH(Metrics::viewport.x, Metrics::viewport.y);
502
503 if (text_dirty || f.id != cachedFontId ||
504 fs != cachedFs || sp != cachedSp || availW != cachedWrapW ||
505 cachedWrapW < 0.0f)
506 {
507 wrapText(availW, fs, sp, f);
508 cachedFs = fs;
509 cachedSp = sp;
510 cachedWrapW = availW;
511 cachedFontId = f.id;
512 text_dirty = false;
513 }
514
515 float maxW = 0.0f, totalH = 0.0f;
516 for (auto &ln : lines)
517 {
518 maxW = std::max(maxW, ln.size.x);
519 totalH += ln.size.y;
520 }
521 cachedSize = {maxW, totalH};
522 return cachedSize;
