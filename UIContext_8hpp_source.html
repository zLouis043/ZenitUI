221 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
222 {
223 char buf[4];
224 size_t len = 0;
225 if (codepoint <= 0x7Fu)
226 {
227 buf[0] = static_cast<char>(codepoint);
228 len = 1;
229 }
230 else if (codepoint <= 0x7FFu)
231 {
232 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
233 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
234 len = 2;
235 }
236 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
237 {
238 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
239 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
240 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
241 len = 3;
242 }
243 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
244 {
245 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
246 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
247 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
248 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
249 len = 4;
250 }
251 if (len == 0)
252 return {0.0f, size};
253 return measureText(f, std::string_view(buf, len), size, spacing);
254 }

595 class TargetPool
596 {
597 public:
598 static constexpr int BUCKET_SIZE = 64;
599 static constexpr int MAX_DIM = 4096;
600
601 static int bucketDim(int v)
602 {
603 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
604 return rounded < MAX_DIM ? rounded : MAX_DIM;
605 }
606
607 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
608 {
609 if (w <= 0 || h <= 0)
610 return {};
611
612 const bool useSubRect = r.supports(Feature::TargetSubRect);
613 const int reqW = useSubRect ? bucketDim(w) : w;
614 const int reqH = useSubRect ? bucketDim(h) : h;
615
616 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
617 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
618 {
619 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
620 {
621 it->inUse = true;
622 it->lastUsedFrame = frameIndex_;
623 return it->handle;
624 }
625 }
626
627 TargetHandle hnd = r.createTarget(reqW, reqH);
628 if (!hnd.valid())
629 return {};
630
631 entries_.push_back({hnd, true, frameIndex_});
632 if (stats)
633 {
634 stats->targetsCreated++;
635 stats->liveTargets++;
636 }
637 return hnd;
638 }
639
640 void release(TargetHandle t)
641 {
642 if (!t.valid())
643 return;
644 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
645 {
646 if (it->handle.id == t.id)
647 {
648 it->inUse = false;
649 it->lastUsedFrame = frameIndex_;
650 return;
651 }
652 }
653 }
654
655 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
656 {
657 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
658 for (auto it = entries_.begin(); it != entries_.end();)
659 {
660 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
661 {
662 r.destroyTarget(it->handle);
663 if (stats)
664 {
665 stats->targetsDestroyed++;
666 if (stats->liveTargets > 0)
667 stats->liveTargets--;
668 }
669 it = entries_.erase(it);
670 }
671 else
672 {
673 ++it;
674 }
675 }
676 frameIndex_++;
677 }
678
679 void clear(IRenderer &r, RenderStats *stats = nullptr)
680 {
681 for (auto &e : entries_)
682 {
683 if (e.handle.valid())
684 {
685 r.destroyTarget(e.handle);
686 if (stats)
687 {
688 stats->targetsDestroyed++;
689 if (stats->liveTargets > 0)
690 stats->liveTargets--;
691 }
692 }
693 }
694 entries_.clear();
695 frameIndex_ = 0;
696 generation_++;
697 }
698
699 size_t liveCount() const { return entries_.size(); }
700
702 uint64_t generation() const { return generation_; }
703
705 uint64_t frameIndex() const { return frameIndex_; }
706
707 private:
708 struct Entry
709 {
710 TargetHandle handle{};
711 bool inUse{false};
712 uint64_t lastUsedFrame{0};
713 };
714 std::vector<Entry> entries_;
715 uint64_t frameIndex_{0};
716 uint64_t generation_{1};
717 };

820 void beginFrame(float delta_time)
821 {
822 dt = delta_time;
823 time += delta_time;
824 wheelConsumedThisFrame = false;
825 buttonConsumed.fill(false);
826 topmostConsumer = nullptr;
827 hoverTarget = nullptr;
828 releaseTarget = nullptr;
829 cullingClipStack.clear();
830 renderStats.beginFrame();
831 activePortals = std::move(framePortals);
832 framePortals.clear();
833 if (platform)
834 {
835 Vec2 newVp = platform->viewportSize();
836 EdgeInsets newSafe = platform->safeArea();
837 if (newVp.x != viewport.x || newVp.y != viewport.y ||
838 newSafe != safeArea)
839 {
840 viewportGeneration++;
841 }
842 viewport = newVp;
843 safeArea = newSafe;
844 pointer = platform->pointer();
845 modifiers = platform->modifiers();
846 inputEvents = platform->pollInputEvents();
847 dpiScale = platform->dpiScale();
848 if (renderer)
849 {
850 renderer->setDpiScale(dpiScale);
851 }
852 }
853 }
854
856 void endFrame()
857 {

859 targetPool.trim(*renderer, &renderStats);
860 }
