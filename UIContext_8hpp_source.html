222 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
223 {
224 char buf[4];
225 size_t len = 0;
226 if (codepoint <= 0x7Fu)
227 {
228 buf[0] = static_cast<char>(codepoint);
229 len = 1;
230 }
231 else if (codepoint <= 0x7FFu)
232 {
233 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
234 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
235 len = 2;
236 }
237 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
238 {
239 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
240 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
241 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
242 len = 3;
243 }
244 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
245 {
246 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
247 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
248 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
249 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
250 len = 4;
251 }
252 if (len == 0)
253 return {0.0f, size};
254 return measureText(f, std::string_view(buf, len), size, spacing);
255 }

596 class TargetPool
597 {
598 public:
599 static constexpr int BUCKET_SIZE = 64;
600 static constexpr int MAX_DIM = 4096;
601
602 static int bucketDim(int v)
603 {
604 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
605 return rounded < MAX_DIM ? rounded : MAX_DIM;
606 }
607
608 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
609 {
610 if (w <= 0 || h <= 0)
611 return {};
612
613 const bool useSubRect = r.supports(Feature::TargetSubRect);
614 const int reqW = useSubRect ? bucketDim(w) : w;
615 const int reqH = useSubRect ? bucketDim(h) : h;
616
617 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
618 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
619 {
620 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
621 {
622 it->inUse = true;
623 it->lastUsedFrame = frameIndex_;
624 return it->handle;
625 }
626 }
627
628 TargetHandle hnd = r.createTarget(reqW, reqH);
629 if (!hnd.valid())
630 return {};
631
632 entries_.push_back({hnd, true, frameIndex_});
633 if (stats)
634 {
635 stats->targetsCreated++;
636 stats->liveTargets++;
637 }
638 return hnd;
639 }
640
641 void release(TargetHandle t)
642 {
643 if (!t.valid())
644 return;
645 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
646 {
647 if (it->handle.id == t.id)
648 {
649 it->inUse = false;
650 it->lastUsedFrame = frameIndex_;
651 return;
652 }
653 }
654 }
655
656 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
657 {
658 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
659 for (auto it = entries_.begin(); it != entries_.end();)
660 {
661 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
662 {
663 r.destroyTarget(it->handle);
664 if (stats)
665 {
666 stats->targetsDestroyed++;
667 if (stats->liveTargets > 0)
668 stats->liveTargets--;
669 }
670 it = entries_.erase(it);
671 }
672 else
673 {
674 ++it;
675 }
676 }
677 frameIndex_++;
678 }
679
680 void clear(IRenderer &r, RenderStats *stats = nullptr)
681 {
682 for (auto &e : entries_)
683 {
684 if (e.handle.valid())
685 {
686 r.destroyTarget(e.handle);
687 if (stats)
688 {
689 stats->targetsDestroyed++;
690 if (stats->liveTargets > 0)
691 stats->liveTargets--;
692 }
693 }
694 }
695 entries_.clear();
696 frameIndex_ = 0;
697 generation_++;
698 }
699
700 size_t liveCount() const { return entries_.size(); }
701
703 uint64_t generation() const { return generation_; }
704
706 uint64_t frameIndex() const { return frameIndex_; }
707
708 private:
709 struct Entry
710 {
711 TargetHandle handle{};
712 bool inUse{false};
713 uint64_t lastUsedFrame{0};
714 };
715 std::vector<Entry> entries_;
716 uint64_t frameIndex_{0};
717 uint64_t generation_{1};
718 };
