223 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
224 {
225 char buf[4];
226 size_t len = 0;
227 if (codepoint <= 0x7Fu)
228 {
229 buf[0] = static_cast<char>(codepoint);
230 len = 1;
231 }
232 else if (codepoint <= 0x7FFu)
233 {
234 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
235 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
236 len = 2;
237 }
238 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
239 {
240 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
241 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
242 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
243 len = 3;
244 }
245 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
246 {
247 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
248 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
249 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
250 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
251 len = 4;
252 }
253 if (len == 0)
254 return {0.0f, size};
255 return measureText(f, std::string_view(buf, len), size, spacing);
256 }

597 class TargetPool
598 {
599 public:
600 static constexpr int BUCKET_SIZE = 64;
601 static constexpr int MAX_DIM = 4096;
602
603 static int bucketDim(int v)
604 {
605 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
606 return rounded < MAX_DIM ? rounded : MAX_DIM;
607 }
608
609 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
610 {
611 if (w <= 0 || h <= 0)
612 return {};
613
614 const bool useSubRect = r.supports(Feature::TargetSubRect);
615 const int reqW = useSubRect ? bucketDim(w) : w;
616 const int reqH = useSubRect ? bucketDim(h) : h;
617
618 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
619 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
620 {
621 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
622 {
623 it->inUse = true;
624 it->lastUsedFrame = frameIndex_;
625 return it->handle;
626 }
627 }
628
629 TargetHandle hnd = r.createTarget(reqW, reqH);
630 if (!hnd.valid())
631 return {};
632
633 entries_.push_back({hnd, true, frameIndex_});
634 if (stats)
635 {
636 stats->targetsCreated++;
637 stats->liveTargets++;
638 }
639 return hnd;
640 }
641
642 void release(TargetHandle t)
643 {
644 if (!t.valid())
645 return;
646 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
647 {
648 if (it->handle.id == t.id)
649 {
650 it->inUse = false;
651 it->lastUsedFrame = frameIndex_;
652 return;
653 }
654 }
655 }
656
657 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
658 {
659 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
660 for (auto it = entries_.begin(); it != entries_.end();)
661 {
662 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
663 {
664 r.destroyTarget(it->handle);
665 if (stats)
666 {
667 stats->targetsDestroyed++;
668 if (stats->liveTargets > 0)
669 stats->liveTargets--;
670 }
671 it = entries_.erase(it);
672 }
673 else
674 {
675 ++it;
676 }
677 }
678 frameIndex_++;
679 }
680
681 void clear(IRenderer &r, RenderStats *stats = nullptr)
682 {
683 for (auto &e : entries_)
684 {
685 if (e.handle.valid())
686 {
687 r.destroyTarget(e.handle);
688 if (stats)
689 {
690 stats->targetsDestroyed++;
691 if (stats->liveTargets > 0)
692 stats->liveTargets--;
693 }
694 }
695 }
696 entries_.clear();
697 frameIndex_ = 0;
698 generation_++;
699 }
700
701 size_t liveCount() const { return entries_.size(); }
702
704 uint64_t generation() const { return generation_; }
705
707 uint64_t frameIndex() const { return frameIndex_; }
708
709 private:
710 struct Entry
711 {
712 TargetHandle handle{};
713 bool inUse{false};
714 uint64_t lastUsedFrame{0};
715 };
716 std::vector<Entry> entries_;
717 uint64_t frameIndex_{0};
718 uint64_t generation_{1};
719 };
