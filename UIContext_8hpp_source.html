107 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
108 {
109 char buf[4];
110 size_t len = 0;
111 if (codepoint <= 0x7Fu)
112 {
113 buf[0] = static_cast<char>(codepoint);
114 len = 1;
115 }
116 else if (codepoint <= 0x7FFu)
117 {
118 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
119 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
120 len = 2;
121 }
122 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
123 {
124 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
125 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
126 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
127 len = 3;
128 }
129 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
130 {
131 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
132 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
133 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
134 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
135 len = 4;
136 }
137 if (len == 0)
138 return {0.0f, size};
139 return measureText(f, std::string_view(buf, len), size, spacing);
140 }

282 class TargetPool
283 {
284 public:
285 static constexpr int BUCKET_SIZE = 64;
286 static constexpr int MAX_DIM = 4096;
287
288 static int bucketDim(int v)
289 {
290 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
291 return rounded < MAX_DIM ? rounded : MAX_DIM;
292 }
293
294 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
295 {
296 if (w <= 0 || h <= 0)
297 return {};
298
299 const bool useSubRect = r.supports(Feature::TargetSubRect);
300 const int reqW = useSubRect ? bucketDim(w) : w;
301 const int reqH = useSubRect ? bucketDim(h) : h;
302
303 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
304 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
305 {
306 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
307 {
308 it->inUse = true;
309 it->lastUsedFrame = frameIndex_;
310 return it->handle;
311 }
312 }
313
314 TargetHandle hnd = r.createTarget(reqW, reqH);
315 if (!hnd.valid())
316 return {};
317
318 entries_.push_back({hnd, true, frameIndex_});
319 if (stats)
320 {
321 stats->targetsCreated++;
322 stats->liveTargets++;
323 }
324 return hnd;
325 }
326
327 void release(TargetHandle t)
328 {
329 if (!t.valid())
330 return;
331 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
332 {
333 if (it->handle.id == t.id)
334 {
335 it->inUse = false;
336 it->lastUsedFrame = frameIndex_;
337 return;
338 }
339 }
340 }
341
342 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
343 {
344 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
345 for (auto it = entries_.begin(); it != entries_.end();)
346 {
347 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
348 {
349 r.destroyTarget(it->handle);
350 if (stats)
351 {
352 stats->targetsDestroyed++;
353 if (stats->liveTargets > 0)
354 stats->liveTargets--;
355 }
356 it = entries_.erase(it);
357 }
358 else
359 {
360 ++it;
361 }
362 }
363 frameIndex_++;
364 }
365
366 void clear(IRenderer &r, RenderStats *stats = nullptr)
367 {
368 for (auto &e : entries_)
369 {
370 if (e.handle.valid())
371 {
372 r.destroyTarget(e.handle);
373 if (stats)
374 {
375 stats->targetsDestroyed++;
376 if (stats->liveTargets > 0)
377 stats->liveTargets--;
378 }
379 }
380 }
381 entries_.clear();
382 frameIndex_ = 0;
383 }
384
385 size_t liveCount() const { return entries_.size(); }
386
387 private:

391 bool inUse{false};
392 uint64_t lastUsedFrame{0};
393 };
394 std::vector<Entry> entries_;
395 uint64_t frameIndex_{0};
