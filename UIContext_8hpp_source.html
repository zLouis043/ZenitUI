516 class TargetPool
517 {
518 public:
519 static constexpr int BUCKET_SIZE = 64;
520 static constexpr int MAX_DIM = 4096;
521
522 static int bucketDim(int v)
523 {
524 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
525 return rounded < MAX_DIM ? rounded : MAX_DIM;
526 }
527
528 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
529 {
530 if (w <= 0 || h <= 0)
531 return {};
532
533 const bool useSubRect = r.supports(Feature::TargetSubRect);
534 const int reqW = useSubRect ? bucketDim(w) : w;
535 const int reqH = useSubRect ? bucketDim(h) : h;
536
537 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
538 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
539 {
540 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
541 {
542 it->inUse = true;
543 it->lastUsedFrame = frameIndex_;
544 return it->handle;
545 }
546 }
547
548 TargetHandle hnd = r.createTarget(reqW, reqH);
549 if (!hnd.valid())
550 return {};
551
552 entries_.push_back({hnd, true, frameIndex_});
553 if (stats)
554 {
555 stats->targetsCreated++;
556 stats->liveTargets++;
557 }
558 return hnd;
559 }
560
561 void release(TargetHandle t)
562 {
563 if (!t.valid())
564 return;
565 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
566 {
567 if (it->handle.id == t.id)
568 {
569 it->inUse = false;
570 it->lastUsedFrame = frameIndex_;
571 return;
572 }
573 }
574 }
575
576 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
577 {
578 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
579 for (auto it = entries_.begin(); it != entries_.end();)
580 {
581 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
582 {
583 r.destroyTarget(it->handle);
584 if (stats)
585 {
586 stats->targetsDestroyed++;
587 if (stats->liveTargets > 0)
588 stats->liveTargets--;
589 }
590 it = entries_.erase(it);
591 }
592 else
593 {
594 ++it;
595 }
596 }
597 frameIndex_++;
598 }
599
600 void clear(IRenderer &r, RenderStats *stats = nullptr)
601 {
602 for (auto &e : entries_)
603 {
604 if (e.handle.valid())
605 {
606 r.destroyTarget(e.handle);
607 if (stats)
608 {
609 stats->targetsDestroyed++;
610 if (stats->liveTargets > 0)
611 stats->liveTargets--;
612 }
613 }
614 }
615 entries_.clear();
616 frameIndex_ = 0;
617 generation_++;
618 }
619
620 size_t liveCount() const { return entries_.size(); }
621
623 uint64_t generation() const { return generation_; }
624
626 uint64_t frameIndex() const { return frameIndex_; }

634 };

637 uint64_t generation_{1};
