210 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
211 {
212 char buf[4];
213 size_t len = 0;
214 if (codepoint <= 0x7Fu)
215 {
216 buf[0] = static_cast<char>(codepoint);
217 len = 1;
218 }
219 else if (codepoint <= 0x7FFu)
220 {
221 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
222 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
223 len = 2;
224 }
225 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
226 {
227 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
228 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
229 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
230 len = 3;
231 }
232 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
233 {
234 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
235 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
236 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
237 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
238 len = 4;
239 }
240 if (len == 0)
241 return {0.0f, size};
242 return measureText(f, std::string_view(buf, len), size, spacing);
243 }

575 class TargetPool
576 {
577 public:
578 static constexpr int BUCKET_SIZE = 64;
579 static constexpr int MAX_DIM = 4096;
580
581 static int bucketDim(int v)
582 {
583 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
584 return rounded < MAX_DIM ? rounded : MAX_DIM;
585 }
586
587 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
588 {
589 if (w <= 0 || h <= 0)
590 return {};
591
592 const bool useSubRect = r.supports(Feature::TargetSubRect);
593 const int reqW = useSubRect ? bucketDim(w) : w;
594 const int reqH = useSubRect ? bucketDim(h) : h;
595
596 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
597 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
598 {
599 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
600 {
601 it->inUse = true;
602 it->lastUsedFrame = frameIndex_;
603 return it->handle;
604 }
605 }
606
607 TargetHandle hnd = r.createTarget(reqW, reqH);
608 if (!hnd.valid())
609 return {};
610
611 entries_.push_back({hnd, true, frameIndex_});
612 if (stats)
613 {
614 stats->targetsCreated++;
615 stats->liveTargets++;
616 }
617 return hnd;
618 }
619
620 void release(TargetHandle t)
621 {
622 if (!t.valid())
623 return;
624 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
625 {
626 if (it->handle.id == t.id)
627 {
628 it->inUse = false;
629 it->lastUsedFrame = frameIndex_;
630 return;
631 }
632 }
633 }
634
635 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
636 {
637 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
638 for (auto it = entries_.begin(); it != entries_.end();)
639 {
640 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
641 {
642 r.destroyTarget(it->handle);
643 if (stats)
644 {
645 stats->targetsDestroyed++;
646 if (stats->liveTargets > 0)
647 stats->liveTargets--;
648 }
649 it = entries_.erase(it);
650 }
651 else
652 {
653 ++it;
654 }
655 }
656 frameIndex_++;
657 }
658
659 void clear(IRenderer &r, RenderStats *stats = nullptr)
660 {
661 for (auto &e : entries_)
662 {
663 if (e.handle.valid())
664 {
665 r.destroyTarget(e.handle);
666 if (stats)
667 {
668 stats->targetsDestroyed++;
669 if (stats->liveTargets > 0)
670 stats->liveTargets--;
671 }
672 }
673 }
674 entries_.clear();
675 frameIndex_ = 0;
676 }
677
678 size_t liveCount() const { return entries_.size(); }
679
680 private:
681 struct Entry
682 {
683 TargetHandle handle{};
684 bool inUse{false};
685 uint64_t lastUsedFrame{0};
686 };
687 std::vector<Entry> entries_;
688 uint64_t frameIndex_{0};
689 };
