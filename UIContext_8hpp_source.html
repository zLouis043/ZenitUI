167 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
168 {
169 char buf[4];
170 size_t len = 0;
171 if (codepoint <= 0x7Fu)
172 {
173 buf[0] = static_cast<char>(codepoint);
174 len = 1;
175 }
176 else if (codepoint <= 0x7FFu)
177 {
178 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
179 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
180 len = 2;
181 }
182 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
183 {
184 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
185 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
186 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
187 len = 3;
188 }
189 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
190 {
191 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
192 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
193 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
194 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
195 len = 4;
196 }
197 if (len == 0)
198 return {0.0f, size};
199 return measureText(f, std::string_view(buf, len), size, spacing);
200 }

532 class TargetPool
533 {
534 public:
535 static constexpr int BUCKET_SIZE = 64;
536 static constexpr int MAX_DIM = 4096;
537
538 static int bucketDim(int v)
539 {
540 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
541 return rounded < MAX_DIM ? rounded : MAX_DIM;
542 }
543
544 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
545 {
546 if (w <= 0 || h <= 0)
547 return {};
548
549 const bool useSubRect = r.supports(Feature::TargetSubRect);
550 const int reqW = useSubRect ? bucketDim(w) : w;
551 const int reqH = useSubRect ? bucketDim(h) : h;
552
553 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
554 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
555 {
556 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
557 {
558 it->inUse = true;
559 it->lastUsedFrame = frameIndex_;
560 return it->handle;
561 }
562 }
563
564 TargetHandle hnd = r.createTarget(reqW, reqH);
565 if (!hnd.valid())
566 return {};
567
568 entries_.push_back({hnd, true, frameIndex_});
569 if (stats)
570 {
571 stats->targetsCreated++;
572 stats->liveTargets++;
573 }
574 return hnd;
575 }
576
577 void release(TargetHandle t)
578 {
579 if (!t.valid())
580 return;
581 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
582 {
583 if (it->handle.id == t.id)
584 {
585 it->inUse = false;
586 it->lastUsedFrame = frameIndex_;
587 return;
588 }
589 }
590 }
591
592 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
593 {
594 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
595 for (auto it = entries_.begin(); it != entries_.end();)
596 {
597 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
598 {
599 r.destroyTarget(it->handle);
600 if (stats)
601 {
602 stats->targetsDestroyed++;
603 if (stats->liveTargets > 0)
604 stats->liveTargets--;
605 }
606 it = entries_.erase(it);
607 }
608 else
609 {
610 ++it;
611 }
612 }
613 frameIndex_++;
614 }
615
616 void clear(IRenderer &r, RenderStats *stats = nullptr)
617 {
618 for (auto &e : entries_)
619 {
620 if (e.handle.valid())
621 {
622 r.destroyTarget(e.handle);
623 if (stats)
624 {
625 stats->targetsDestroyed++;
626 if (stats->liveTargets > 0)
627 stats->liveTargets--;
628 }
629 }
630 }
631 entries_.clear();
632 frameIndex_ = 0;
633 }
634
635 size_t liveCount() const { return entries_.size(); }
636
637 private:
638 struct Entry
639 {
640 TargetHandle handle{};
641 bool inUse{false};
642 uint64_t lastUsedFrame{0};
643 };
644 std::vector<Entry> entries_;
645 uint64_t frameIndex_{0};
646 };
