152 virtual Vec2 measureCodepoint(FontHandle f, uint32_t codepoint, float size, float spacing)
153 {
154 char buf[4];
155 size_t len = 0;
156 if (codepoint <= 0x7Fu)
157 {
158 buf[0] = static_cast<char>(codepoint);
159 len = 1;
160 }
161 else if (codepoint <= 0x7FFu)
162 {
163 buf[0] = static_cast<char>(0xC0u | (codepoint >> 6));
164 buf[1] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
165 len = 2;
166 }
167 else if (codepoint <= 0xFFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
168 {
169 buf[0] = static_cast<char>(0xE0u | (codepoint >> 12));
170 buf[1] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
171 buf[2] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
172 len = 3;
173 }
174 else if (codepoint <= 0x10FFFFu && !(codepoint >= 0xD800u && codepoint <= 0xDFFFu))
175 {
176 buf[0] = static_cast<char>(0xF0u | (codepoint >> 18));
177 buf[1] = static_cast<char>(0x80u | ((codepoint >> 12) & 0x3Fu));
178 buf[2] = static_cast<char>(0x80u | ((codepoint >> 6) & 0x3Fu));
179 buf[3] = static_cast<char>(0x80u | (codepoint & 0x3Fu));
180 len = 4;
181 }
182 if (len == 0)
183 return {0.0f, size};
184 return measureText(f, std::string_view(buf, len), size, spacing);
185 }

472 class TargetPool
473 {
474 public:
475 static constexpr int BUCKET_SIZE = 64;
476 static constexpr int MAX_DIM = 4096;
477
478 static int bucketDim(int v)
479 {
480 int rounded = ((v + BUCKET_SIZE - 1) / BUCKET_SIZE) * BUCKET_SIZE;
481 return rounded < MAX_DIM ? rounded : MAX_DIM;
482 }
483
484 TargetHandle acquire(IRenderer &r, int w, int h, RenderStats *stats = nullptr)
485 {
486 if (w <= 0 || h <= 0)
487 return {};
488
489 const bool useSubRect = r.supports(Feature::TargetSubRect);
490 const int reqW = useSubRect ? bucketDim(w) : w;
491 const int reqH = useSubRect ? bucketDim(h) : h;
492
493 // Ricerca LIFO di un target libero con le dimensioni richieste/arrotondate.
494 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
495 {
496 if (!it->inUse && it->handle.width == reqW && it->handle.height == reqH)
497 {
498 it->inUse = true;
499 it->lastUsedFrame = frameIndex_;
500 return it->handle;
501 }
502 }
503
504 TargetHandle hnd = r.createTarget(reqW, reqH);
505 if (!hnd.valid())
506 return {};
507
508 entries_.push_back({hnd, true, frameIndex_});
509 if (stats)
510 {
511 stats->targetsCreated++;
512 stats->liveTargets++;
513 }
514 return hnd;
515 }
516
517 void release(TargetHandle t)
518 {
519 if (!t.valid())
520 return;
521 for (auto it = entries_.rbegin(); it != entries_.rend(); ++it)
522 {
523 if (it->handle.id == t.id)
524 {
525 it->inUse = false;
526 it->lastUsedFrame = frameIndex_;
527 return;
528 }
529 }
530 }
531
532 void trim(IRenderer &r, RenderStats *stats = nullptr, int graceFrames = 120)
533 {
534 const uint64_t grace = graceFrames > 0 ? static_cast<uint64_t>(graceFrames) : 1u;
535 for (auto it = entries_.begin(); it != entries_.end();)
536 {
537 if (!it->inUse && (frameIndex_ - it->lastUsedFrame) >= grace)
538 {
539 r.destroyTarget(it->handle);
540 if (stats)
541 {
542 stats->targetsDestroyed++;
543 if (stats->liveTargets > 0)
544 stats->liveTargets--;
545 }
546 it = entries_.erase(it);
547 }
548 else
549 {
550 ++it;
551 }
552 }
553 frameIndex_++;
554 }
555
556 void clear(IRenderer &r, RenderStats *stats = nullptr)
557 {
558 for (auto &e : entries_)
559 {
560 if (e.handle.valid())
561 {
562 r.destroyTarget(e.handle);
563 if (stats)
564 {
565 stats->targetsDestroyed++;
566 if (stats->liveTargets > 0)
567 stats->liveTargets--;
568 }
569 }
570 }
571 entries_.clear();
572 frameIndex_ = 0;
573 }
574
575 size_t liveCount() const { return entries_.size(); }

583 };
