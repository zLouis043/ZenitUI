73 inline std::string valueToString(const Value &v)
74 {
75 if (v.isAuto())
76 return "auto";
77 std::ostringstream os;
78 for (size_t i = 0; i < v.terms.size(); ++i)
79 {
80 const auto &t = v.terms[i];
81 if (i == 0)
82 {
83 if (t.coeff < 0.0f)
84 os << "-";
85 os << std::abs(t.coeff) << unitName(t.unit);
86 }
87 else
88 {
89 os << (t.coeff < 0.0f ? " - " : " + ");
90 os << std::abs(t.coeff) << unitName(t.unit);
91 }
92 }
93 return os.str();
94 }

416 {
417 std::ostringstream os;
418 for (size_t i = 0; i < r.chain.size(); ++i)
419 {
420 if (i > 0)
421 os << " ";

423
424 auto printComponent = [&](SimpleSelector::Kind kind,
425 const std::string &name)
426 {
427 if (kind == SimpleSelector::Kind::Class)
428 os << ".";
429 else if (kind == SimpleSelector::Kind::Id)
430 os << "#";
431 os << name;
432 };
433
434 printComponent(ss.kind, ss.name);
435 for (const auto &[kind, name] : ss.extras)
436 printComponent(kind, name);
437
438 if (ss.requireHover)
439 os << ":hover";
440 if (ss.requirePressed)
441 os << ":pressed";
442 if (ss.requireFocus)
443 os << ":focus";
444 if (ss.requireDisabled)
445 os << ":disabled";

447 os << ":checked";
448 }
449 if (!r.part.empty())
450 os << "::" << r.part;
451 return os.str();

474 inline void dumpStyle(const Layout &node, std::ostream &os = std::cerr)
475 {
476 os << "=== Node ===\n";
477 os << " tag : " << node.getStyleTag() << "\n";
478 if (!node.nodeId.empty())
479 os << " id : #" << node.nodeId << "\n";
480 if (!node.ref.empty())
481 os << " ref : @" << node.ref << "\n";
482 if (!node.getStyleClasses().empty())
483 {
484 os << " classes :";
485 for (const auto &c : node.getStyleClasses())
486 os << " ." << c;
487 os << "\n";
488 }
489 os << " state : " << stateFlagsToString(node) << "\n";
490 const auto &r = node.getRect();

492 os << " style : " << computedStyleToString(node.getStyle()) << "\n";
493
494 auto matches = node.getMatchingRules();
495 os << " matches : " << matches.size() << " rule(s)\n";
496 for (const auto *rule : matches)
497 {
498 os << "spec=" << rule->specificity.ids << "."
499 << rule->specificity.classes << "."
500 << rule->specificity.tags << "] "
501 << selectorToString(*rule) << "\n";
502 os << " -> { " << styleToString(rule->style) << " }\n";
503 }

510 inline void dumpTree(const Layout &node, std::ostream &os = std::cerr,
511 int maxDepth = -1, int depth = 0)
512 {
513 std::string indent(depth * 2, ' ');
514 os << indent << "- ";
515 if (!node.nodeId.empty())
516 os << "#" << node.nodeId << " ";
517 if (!node.ref.empty())
518 os << "@" << node.ref << " ";
519 os << node.getStyleTag();
520 for (const auto &c : node.getStyleClasses())
521 os << "." << c;
522
523 const auto &r = node.getRect();
524 os << " [" << r.width << "x" << r.height << " @ " << r.x << "," << r.y << "]";
525
526 std::string st = stateFlagsToString(node);
527 if (st != "-")
528 os << " {" << st << "}";
529 os << "\n";
530
531 if (maxDepth >= 0 && depth >= maxDepth)
532 return;
533 for (const auto &c : node.children)
534 dumpTree(*c, os, maxDepth, depth + 1);
535 }

574 inline void dumpParts(const Layout &node, std::ostream &os = std::cerr)
575 {
576 static const char *candidates[] = {
577 "track", "fill", "knob", "box", "mark", "cursor"};
578 os << "=== Parts of " << node.getStyleTag() << " ===\n";
579 for (auto *name : candidates)
580 {
581 auto &theme = Theme::get();
582 int matchCount = 0;
583 for (const auto &r : theme.rules)
584 {
585 if (r.part != name)
586 continue;
587 if (ruleMatches(r, &node))
588 matchCount++;
589 }
590 if (matchCount > 0)
591 {
592 os << " ::" << name << " (" << matchCount << " matching rule(s))\n";
593 }
