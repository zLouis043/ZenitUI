679 static std::vector<VisualLine> wrapString(const std::string &src, float availW,
680 FontHandle f, float fs, float sp)
681 {
682 std::vector<VisualLine> out;
683 auto r = UIContext::get().renderer;
684 auto measure = [&](size_t a, size_t b) -> float
685 {
686 if (!r || b <= a)
687 return 0.0f;
688 return r->measureText(f, std::string_view(src).substr(a, b - a), fs, sp).x;
689 };
690
691 size_t len = src.size();
692 size_t pStart = 0;
693 while (true)
694 {
695 size_t pEnd = pStart;
696 while (pEnd < len && src[pEnd] != '\n')
697 ++pEnd;
698
699 if (pStart == pEnd || availW <= 1.0f || !r)
700 {
701 out.push_back({static_cast<int>(pStart), static_cast<int>(pEnd), true});
702 }
703 else
704 {
705 auto bounds = Utf8::clusterBoundaries(std::string_view(src).substr(pStart, pEnd - pStart));
706 size_t lineStartIdx = 0;
707 const size_t lastIdx = bounds.size() - 1;
708 while (lineStartIdx < lastIdx)
709 {
710 size_t lineStart = pStart + bounds[lineStartIdx];
711 if (measure(lineStart, pEnd) <= availW)
712 {
713 out.push_back({static_cast<int>(lineStart), static_cast<int>(pEnd), true});
714 break;
715 }
716
717 size_t lo = lineStartIdx + 1;
718 size_t hi = lastIdx;
719 while (lo < hi)
720 {
721 size_t mid = lo + (hi - lo + 1) / 2;
722 if (measure(lineStart, pStart + bounds[mid]) <= availW)
723 lo = mid;
724 else
725 hi = mid - 1;
726 }
727 size_t fitIdx = std::max(lineStartIdx + 1, lo);
728 size_t breakIdx = fitIdx;
729
730 if (fitIdx < lastIdx && src[pStart + bounds[fitIdx]] == ' ')
731 {
732 breakIdx = fitIdx + 1;
733 }
734 else
735 {
736 for (size_t k = fitIdx - 1; k > lineStartIdx; --k)
737 {
738 if (src[pStart + bounds[k]] == ' ')
739 {
740 breakIdx = k + 1;
741 break;
742 }
743 }
744 }
745
746 size_t breakPos = pStart + bounds[breakIdx];
747 out.push_back({static_cast<int>(lineStart), static_cast<int>(breakPos), false});
748 lineStartIdx = breakIdx;
749 }
750 }
751
752 if (pEnd >= len)
753 break;
754 pStart = pEnd + 1;
755 }
756
757 if (out.empty())
758 out.push_back({0, 0, true});
759 return out;
760 }

794 int closestIndexInLine(int lineIdx, float targetX, FontHandle f, float fs, float sp) const
795 {
796 auto r = UIContext::get().renderer;
797 const auto &ln = visualLines[lineIdx];
798 if (!r)
799 return ln.start;
800
801 int maxIdx = ln.hardBreak
802 ? ln.end
803 : std::max(ln.start,
804 static_cast<int>(Utf8::prevCluster(text, static_cast<size_t>(ln.end))));
805 if (maxIdx <= ln.start)
806 return ln.start;
807
808 auto bounds = Utf8::clusterBoundaries(
809 std::string_view(text).substr(static_cast<size_t>(ln.start),
810 static_cast<size_t>(maxIdx - ln.start)));
811
812 int best = ln.start;
813 float bestDist = std::abs(targetX);
814 for (size_t bi = 1; bi < bounds.size(); ++bi)
815 {
816 size_t bytePos = static_cast<size_t>(ln.start) + bounds[bi];
817 float w = r->measureText(
818 f,
819 std::string_view(text).substr(static_cast<size_t>(ln.start), bounds[bi]),
820 fs, sp)
821 .x;
822 float dist = std::abs(targetX - w);
823 if (dist < bestDist)
824 {
825 bestDist = dist;
826 best = static_cast<int>(bytePos);
827 }
828 }
829 return best;
830 }

888 void selectWordAtCursor()
889 {
890 if (text.empty())
891 return;
892 size_t s = Utf8::snapFloor(text, static_cast<size_t>(std::max(0, cursorPos)));
893 size_t e = s;
894 while (s > 0)
895 {
896 size_t prev = Utf8::prevCluster(text, s);
897 uint32_t cp = 0;
898 Utf8::decode(text, prev, cp);
899 if (!Utf8::isWordChar(cp))
900 break;
901 s = prev;
902 }
903 while (e < text.size())
904 {
905 uint32_t cp = 0;
906 Utf8::decode(text, e, cp);
907 if (!Utf8::isWordChar(cp))
908 break;
909 e = Utf8::nextCluster(text, e);
910 }
911 if (s != e)
912 {
913 selectionStart = static_cast<int>(s);
914 cursorPos = static_cast<int>(e);
915 }
916 }
