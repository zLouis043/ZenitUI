235 inline void VirtualView::rebuildCumulative(float availW)
236 {
237 cumulativeHeights.assign((size_t)itemCount + 1, 0.0f);
238
239 if (uniformHeight > 0.0f)
240 {
241 for (int i = 0; i <= itemCount; ++i)
242 cumulativeHeights[i] = (float)i * uniformHeight;
243 lastMeasuredWidth = availW;
244 return;
245 }
246
247 float acc = 0.0f;
248 for (int i = 0; i < itemCount; ++i)
249 {
250 std::shared_ptr<Layout> node;
251 if (itemFactory)
252 node = itemFactory(i);
253 if (node)
254 {
255 Vec2 sz = node->measure(availW, 1e9f);
256 acc += std::max(1.0f, sz.y);
257 }
258 else
259 {
260 acc += 1.0f;
261 }
262 cumulativeHeights[i + 1] = acc;
263 }
264 lastMeasuredWidth = availW;
265 }

334 inline void VirtualView::addMaterialized(int index)
335 {
336 if (hasMaterialized(index))
337 return;
338 if (!itemFactory)
339 return;
340 if (cumulativeHeights.size() != (size_t)itemCount + 1)
341 return;
342
343 auto node = itemFactory(index);
344 if (!node)
345 return;
346
347 float top = cumulativeHeights[index];
348 float bot = cumulativeHeights[index + 1];
349
350 node->getInlineBase().position = Position::Absolute;
351 node->getInlineBase().left = Px(0.0f);
352 node->getInlineBase().width = Percent(100.0f);
353 node->getInlineBase().top = Px(top);
354 node->getInlineBase().height = Px(std::max(1.0f, bot - top));
355 node->setEnabled(true);
356
357 addChild(node);
358 materialized.push_back({node, index});
359
360 if (onItemCreated)
361 onItemCreated(node.get(), index);

376 inline void VirtualView::updateMaterialization()
377 {
378 if (itemCount <= 0)
379 {
380 for (auto &e : materialized)
381 e.node->removeFromParent();
382 materialized.clear();
383 return;
384 }
385
386 if (cumulativeHeights.size() != (size_t)itemCount + 1)
387 return;
388
389 int first = getFirstVisibleItem();
390 int last = getLastVisibleItem();
391 if (first < 0 || last < 0)
392 return;
393
394 int start = std::max(0, first - 2);
395 int end = std::min(itemCount - 1, last + 2);
396
397 for (auto it = materialized.begin(); it != materialized.end();)
398 {
399 if (it->index < start || it->index > end)
400 {
401 it->node->removeFromParent();
402 it = materialized.erase(it);
403 }
404 else
405 {
406 ++it;
407 }
408 }
409
410 for (int i = start; i <= end; ++i)
411 addMaterialized(i);
412 }
