240 inline void VirtualView::rebuildCumulative(float availW)
241 {
242 cumulativeHeights.assign((size_t)itemCount + 1, 0.0f);
243
244 if (uniformHeight > 0.0f)
245 {
246 for (int i = 0; i <= itemCount; ++i)
247 cumulativeHeights[i] = (float)i * uniformHeight;
248 lastMeasuredWidth = availW;
249 return;
250 }
251
252 float acc = 0.0f;
253 for (int i = 0; i < itemCount; ++i)
254 {
255 std::shared_ptr<Layout> node;
256 if (itemFactory)
257 node = itemFactory(i);
258 if (node)
259 {
260 Vec2 sz = node->measure(availW, 1e9f);
261 acc += std::max(1.0f, sz.y);
262 }
263 else
264 {
265 acc += 1.0f;
266 }
267 cumulativeHeights[i + 1] = acc;
268 }
269 lastMeasuredWidth = availW;
270 }

339 inline void VirtualView::addMaterialized(int index)
340 {
341 if (hasMaterialized(index))
342 return;
343 if (!itemFactory)
344 return;
345 if (cumulativeHeights.size() != (size_t)itemCount + 1)
346 return;
347
348 auto node = itemFactory(index);
349 if (!node)
350 return;
351
352 float top = cumulativeHeights[index];
353 float bot = cumulativeHeights[index + 1];
354
355 node->getInlineBase().position = Position::Absolute;
356 node->getInlineBase().left = Px(0.0f);
357 node->getInlineBase().width = Percent(100.0f);
358 node->getInlineBase().top = Px(top);
359 node->getInlineBase().height = Px(std::max(1.0f, bot - top));
360 node->setEnabled(true);
361
362 addChild(node);
363 materialized.push_back({node, index});
364
365 if (onItemCreated)
366 onItemCreated(node.get(), index);
367 }

381 inline void VirtualView::updateMaterialization()
382 {
383 if (itemCount <= 0)
384 {
385 for (auto &e : materialized)
386 e.node->removeFromParent();
387 materialized.clear();
388 return;
389 }
390
391 if (cumulativeHeights.size() != (size_t)itemCount + 1)
392 return;
393
394 int first = getFirstVisibleItem();
395 int last = getLastVisibleItem();
396 if (first < 0 || last < 0)
397 return;
398
399 int start = std::max(0, first - 2);
400 int end = std::min(itemCount - 1, last + 2);
401
402 for (auto it = materialized.begin(); it != materialized.end();)
403 {
404 if (it->index < start || it->index > end)
405 {
406 it->node->removeFromParent();
407 it = materialized.erase(it);
408 }
409 else
410 {
411 ++it;
412 }
413 }
414
415 for (int i = start; i <= end; ++i)
416 addMaterialized(i);
417 }

419 inline Vec2 VirtualView::computeIntrinsicSize(float availW, float /*availH*/)
420 {
421 if (cumulativeHeights.size() != (size_t)itemCount + 1 ||
422 (uniformHeight == 0.0f && lastMeasuredWidth != availW))
423 {
424 rebuildCumulative(availW);
425 }
426
427 float maxW = availW;
428 const auto &st = getStyle();
429 if (st.overflowX == Overflow::Auto || st.overflowX == Overflow::Scroll)
430 {
431 for (const auto &e : materialized)
432 {
433 if (!e.node)
434 continue;
435 auto &ib = e.node->getInlineBase();
436 auto prevW = ib.width;
437 ib.width = Auto();
438 Vec2 sz = e.node->measure(1e9f, 1e9f);
439 ib.width = prevW;
440 maxW = std::max(maxW, sz.x);
441 }
442 }
443
444 float totalH = cumulativeHeights.empty() ? 0.0f : cumulativeHeights.back();
445 return {maxW, totalH};
