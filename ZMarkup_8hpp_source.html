288 template <typename T = Layout>
289 std::shared_ptr<T> ref(const std::string &name) const
290 {
291 auto it = ctx_->byRef.find(name);
292 if (it == ctx_->byRef.end() || it->second.empty())
293 {
294 warnOnce("ref:" + name,
295 "UINode::ref('" + name + "'): no node with this ref");
296 return nullptr;
297 }
298 std::shared_ptr<Layout> firstLive;
299 size_t liveCount = 0;
300 for (const auto &wp : it->second)
301 {
302 if (auto sp = wp.lock())
303 {
304 if (!firstLive)
305 firstLive = sp;
306 ++liveCount;
307 }
308 }
309 if (!firstLive)
310 return nullptr;
311 if (liveCount > 1)
312 {
313 warnOnce("ref-ambig:" + name,
314 "UINode::ref('" + name + "'): " +
315 std::to_string(liveCount) +
316 " nodes match, returning the first. "
317 "Use refAll for the full list.");
318 }
319 return std::static_pointer_cast<T>(firstLive);
320 }

323 template <typename T = Layout>
324 std::vector<std::shared_ptr<T>> refAll(const std::string &name) const
325 {
326 auto it = ctx_->byRef.find(name);
327 if (it == ctx_->byRef.end() || it->second.empty())
328 {
329 logDebug("ZMarkup", "", 0, 0,
330 "UINode::refAll('" + name + "'): empty");
331 return {};
332 }
333 std::vector<std::shared_ptr<T>> out;
334 out.reserve(it->second.size());
335 for (const auto &wp : it->second)
336 {
337 if (auto sp = wp.lock())
338 out.push_back(std::static_pointer_cast<T>(sp));
339 }
340 return out;
341 }

379 template <typename Range, typename Fn>
380 UINode &syncEach(const std::string &eachRef, const Range &items, Fn &&binder)
381 {
382 auto it = ctx_->eaches.find(eachRef);
383 if (it == ctx_->eaches.end())
384 {
385 warnOnce("each:" + eachRef,
386 "UINode::syncEach('" + eachRef + "'): no Each with this ref");
387 return *this;
388 }
389
390 auto &binding = it->second;
391 auto anchor = binding.anchor.lock();
392 auto parent = anchor ? anchor->getParent() : nullptr;
393 if (!parent)
394 return *this;
395
396 binding.keyedStore.reset();
397 clearEachNodes_(binding, *parent);
398
399 const bool effVis = ctx_->isConditionVisible(binding.parentCondition);
400 const bool isEmpty = (std::begin(items) == std::end(items));
401
402 if (isEmpty)
403 {
404 mountEachFallback_(binding, *parent, anchor.get(), effVis);
405 return *this;
406 }
407
408 binding.showingFallback = false;
409 size_t idx = 0;
410 for (const auto &elem : items)
411 {
412 UINode itemUi = getGlobalRegistry().instantiate(binding.prefabName);
413 auto itemRoot = itemUi.root();
414 if (!itemRoot)
415 break;
416
417 invokeEachCallback_(binder, itemUi, elem, idx);
418 if (!effVis)
419 itemRoot->setVisible(false);
420
421 parent->insertChildBefore(anchor.get(), itemRoot);
422 binding.spawnedNodes.push_back(itemRoot);
423 ++idx;
424 }
425 return *this;
426 }

429 UINode &bindEach(const std::string &eachRef,

431 Fn binder,
432 bool fireImmediately = true)
433 {
434 if (ctx_->eaches.find(eachRef) == ctx_->eaches.end())
435 {
436 warnOnce("each:" + eachRef,

441 attachToRoot_();
442

462 template <typename Range, typename KeyFn, typename OnCreateFn, typename OnUpdateFn>
463 UINode &syncEachKeyed(const std::string &eachRef,
464 const Range &items,
465 KeyFn &&keyFn,
466 OnCreateFn &&onCreate,
467 OnUpdateFn &&onUpdate)
468 {
469 auto it = ctx_->eaches.find(eachRef);
470 if (it == ctx_->eaches.end())
471 {
472 warnOnce("each:" + eachRef,
473 "UINode::syncEachKeyed('" + eachRef + "'): no Each with this ref");
474 return *this;
475 }
476
477 auto &binding = it->second;
478 auto anchor = binding.anchor.lock();
479 auto parent = anchor ? anchor->getParent() : nullptr;
480 if (!parent)
481 return *this;
482
483 using ElemT = std::decay_t<decltype(*std::begin(items))>;
484 using KeyT = std::decay_t<std::invoke_result_t<KeyFn &, const ElemT &>>;
485
486 const bool effVis = ctx_->isConditionVisible(binding.parentCondition);
487 const bool isEmpty = (std::begin(items) == std::end(items));
488
489 if (isEmpty)
490 {
491 binding.keyedStore.reset();
492 clearEachNodes_(binding, *parent);
493 mountEachFallback_(binding, *parent, anchor.get(), effVis);
494 return *this;
495 }
496
497 if (binding.showingFallback)
498 clearEachNodes_(binding, *parent);
499
500 auto *store = dynamic_cast<KeyedStore<KeyT> *>(binding.keyedStore.get());
501 if (!store)
502 {
503 auto newStore = std::make_shared<KeyedStore<KeyT>>();
504 store = newStore.get();
505 binding.keyedStore = std::move(newStore);
506 }
507
508 std::unordered_map<KeyT, typename KeyedStore<KeyT>::Entry> nextEntries;
509 nextEntries.reserve(store->entries.size() + 4);
510 std::vector<std::shared_ptr<Layout>> nextNodes;
511
512 size_t idx = 0;
513 for (const auto &elem : items)
514 {
515 KeyT key = keyFn(elem);
516 if (nextEntries.count(key) > 0)
517 {
518 logWarn("ZMarkup", "", 0, 0,
519 "UINode::syncEachKeyed('" + eachRef + "'): duplicate key detected, skipping duplicate item");
520 ++idx;
521 continue;
522 }
523
524 auto existing = store->entries.find(key);
525 std::shared_ptr<Layout> existingRoot =
526 (existing != store->entries.end()) ? existing->second.root.lock() : nullptr;
527
528 if (existingRoot)
529 {
530 auto entry = std::move(existing->second);
531 UINode itemUi(existingRoot, entry.ctx);
532 invokeEachCallback_(onUpdate, itemUi, elem, idx);
533 if (!effVis)
534 existingRoot->setVisible(false);
535
536 nextNodes.push_back(std::move(existingRoot));
537 nextEntries.emplace(std::move(key), std::move(entry));
538 }
539 else
540 {
541 UINode itemUi = getGlobalRegistry().instantiate(binding.prefabName);
542 auto itemRoot = itemUi.root();
543 if (!itemRoot)
544 break;
545
546 invokeEachCallback_(onCreate, itemUi, elem, idx);
547 invokeEachCallback_(onUpdate, itemUi, elem, idx);
548 if (!effVis)
549 itemRoot->setVisible(false);
550
551 nextEntries.emplace(std::move(key),
552 typename KeyedStore<KeyT>::Entry{itemRoot, itemUi.context()});
553 nextNodes.push_back(std::move(itemRoot));
554 }
555 ++idx;
556 }
557
558 bool sameOrder = (nextNodes.size() == binding.spawnedNodes.size());
559 if (sameOrder)
560 {
561 for (size_t i = 0; i < nextNodes.size(); ++i)
562 {
563 if (nextNodes[i].get() != binding.spawnedNodes[i].lock().get())
564 {
565 sameOrder = false;
566 break;
567 }
568 }
569 }
570
571 if (!sameOrder)
572 {
573 for (auto &oldWp : binding.spawnedNodes)
574 {
575 if (auto oldNode = oldWp.lock())
576 parent->removeChild(oldNode.get());
577 }
578 for (auto &newNode : nextNodes)
579 {
580 parent->insertChildBefore(anchor.get(), newNode);
581 }
582 }
583
584 store->entries = std::move(nextEntries);
585 binding.spawnedNodes.clear();
586 binding.spawnedNodes.reserve(nextNodes.size());
587 for (const auto &n : nextNodes)
588 binding.spawnedNodes.push_back(n);
589 binding.showingFallback = false;
590 return *this;
591 }
