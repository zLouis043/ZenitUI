489 inline Color lerpProp(const Color &a, const Color &b, float t)
490 {
491 auto mix = [t](uint8_t x, uint8_t y)
492 {
493 float v = static_cast<float>(x) + (static_cast<float>(y) - static_cast<float>(x)) * t;
494 return static_cast<uint8_t>(std::clamp(std::lround(v), 0L, 255L));
495 };
496 return {mix(a.r, b.r), mix(a.g, b.g), mix(a.b, b.b), mix(a.a, b.a)};
497 }

504 inline Value lerpProp(const Value &a, const Value &b, float t)
505 {
506 if (a.isAuto() && b.isAuto())
507 return Value::autoSize();
508 if (a.isAuto() || b.isAuto())
509 return t > 0.0f ? b : a;
510
511 if (a.op != ValueOp::None || b.op != ValueOp::None)
512 {
513 return a + (b - a).mulWith(Value::number(t));
514 }
515
516 float ca[7] = {0}, cb[7] = {0};
517 for (const auto &term : a.terms)
518 {
519 int i = static_cast<int>(term.unit);
520 if (i >= 0 && i < 7)
521 ca[i] += term.coeff;
522 }
523 for (const auto &term : b.terms)
524 {
525 int i = static_cast<int>(term.unit);
526 if (i >= 0 && i < 7)
527 cb[i] += term.coeff;
528 }
529
530 Value r;
531 for (int i = 0; i < 7; ++i)
532 {
533 float v = ca[i] + (cb[i] - ca[i]) * t;
534 if (v != 0.0f)
535 r.terms.push_back({v, static_cast<Unit>(i)});
536 }
537 return r;
538 }

589 {
590 ComputedStyle r;
591
592 auto timingFor = [&](const char *propName) -> std::pair<float, TransitionFunction>
593 {
594 for (const auto &spec : b.transitions)
595 {
596 if (spec.prop == propName || spec.prop == "all")
597 return {spec.duration, spec.ease};
598 }

602#define X(T, name, def) \
