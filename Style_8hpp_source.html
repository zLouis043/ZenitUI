486 inline Color lerpProp(const Color &a, const Color &b, float t)
487 {
488 auto mix = [t](uint8_t x, uint8_t y)
489 {
490 float v = static_cast<float>(x) + (static_cast<float>(y) - static_cast<float>(x)) * t;
491 return static_cast<uint8_t>(std::clamp(std::lround(v), 0L, 255L));
492 };
493 return {mix(a.r, b.r), mix(a.g, b.g), mix(a.b, b.b), mix(a.a, b.a)};
494 }

501 inline Value lerpProp(const Value &a, const Value &b, float t)
502 {
503 if (a.isAuto() && b.isAuto())
504 return Value::autoSize();
505 if (a.isAuto() || b.isAuto())
506 return t > 0.0f ? b : a;
507
508 if (a.op != ValueOp::None || b.op != ValueOp::None)
509 {
510 return a + (b - a).mulWith(Value::number(t));
511 }
512
513 float ca[7] = {0}, cb[7] = {0};
514 for (const auto &term : a.terms)
515 {
516 int i = static_cast<int>(term.unit);
517 if (i >= 0 && i < 7)
518 ca[i] += term.coeff;
519 }
520 for (const auto &term : b.terms)
521 {
522 int i = static_cast<int>(term.unit);
523 if (i >= 0 && i < 7)
524 cb[i] += term.coeff;
525 }
526
527 Value r;
528 for (int i = 0; i < 7; ++i)
529 {
530 float v = ca[i] + (cb[i] - ca[i]) * t;
531 if (v != 0.0f)
532 r.terms.push_back({v, static_cast<Unit>(i)});
533 }
534 return r;
535 }

586 {
587 ComputedStyle r;
588
589 auto timingFor = [&](const char *propName) -> std::pair<float, TransitionFunction>
590 {
591 for (const auto &spec : b.transitions)
592 {
593 if (spec.prop == propName || spec.prop == "all")
594 return {spec.duration, spec.ease};
