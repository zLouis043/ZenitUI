150 inline float sampleActive(const ActiveCssAnimation &a, bool &finished)
151 {
152 finished = false;
153 float t = a.elapsed - a.delay;
154 if (t < 0.0f)
155 return 0.0f;
156 if (a.duration <= 0.0f)
157 {
158 finished = true;
159 return 1.0f;
160 }
161
162 float total = (a.iterations < 0) ? -1.0f : (a.duration * (float)a.iterations);
163 if (a.iterations >= 0 && t >= total)
164 {
165 t = total;
166 finished = true;
167 }
168 float iter = std::floor(t / a.duration);
169
170 if (a.iterations > 0 && finished && iter > 0.0f)
171 {
172 iter = std::min(iter, (float)(a.iterations - 1));
173 }
174
175 float local = (t - iter * a.duration) / a.duration;
176 if (a.alternate && ((int)iter % 2 == 1))
177 local = 1.0f - local;
178 return std::clamp(local, 0.0f, 1.0f);
179 }

207 {
208 Style out;

211 if (anim.keyframes.size() == 1)

216 for (size_t i = 0; i + 1 < anim.keyframes.size(); ++i)

218 if (localT >= anim.keyframes[i].t && localT <= anim.keyframes[i + 1].t)
