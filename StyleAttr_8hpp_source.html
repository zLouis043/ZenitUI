74 inline std::vector<CalcToken> tokenizeCalc(std::string_view s)
75 {
76 std::vector<CalcToken> out;
77 size_t i = 0;
78 while (i < s.size())
79 {
80 char c = s[i];
81 if (std::isspace((unsigned char)c))
82 {
83 ++i;
84 continue;
85 }
86 if (c == '+')
87 {
88 out.push_back({CalcToken::Kind::Plus});
89 ++i;
90 continue;
91 }
92 if (c == '-')
93 {
94 out.push_back({CalcToken::Kind::Minus});
95 ++i;
96 continue;
97 }
98 if (c == '*')
99 {
100 out.push_back({CalcToken::Kind::Star});
101 ++i;
102 continue;
103 }
104 if (c == '/')
105 {
106 out.push_back({CalcToken::Kind::Slash});
107 ++i;
108 continue;
109 }
110 if (c == '(')
111 {
112 out.push_back({CalcToken::Kind::LParen});
113 ++i;
114 continue;
115 }
116 if (c == ')')
117 {
118 out.push_back({CalcToken::Kind::RParen});
119 ++i;
120 continue;
121 }
122 if (c == '%')
123 {
124 out.push_back({CalcToken::Kind::Percent});
125 ++i;
126 continue;
127 }
128 if (c == ',')
129 {
130 out.push_back({CalcToken::Kind::Comma});
131 ++i;
132 continue;
133 }
134
135 if (std::isdigit((unsigned char)c) || c == '.')
136 {
137 size_t start = i;
138 while (i < s.size() && (std::isdigit((unsigned char)s[i]) || s[i] == '.'))
139 ++i;
140 float v = 0.0f;
141 (void)parseFloatSV(s.substr(start, i - start), v);
142 CalcToken t;
143 t.kind = CalcToken::Kind::Number;
144 t.num = v;
145 out.push_back(t);
146 continue;
147 }
148 if (std::isalpha((unsigned char)c) || c == '_')
149 {
150 size_t start = i;
151 while (i < s.size() && (std::isalnum((unsigned char)s[i]) || s[i] == '_'))
152 ++i;
153 CalcToken t;
154 t.kind = CalcToken::Kind::Ident;
155 t.ident = std::string(s.substr(start, i - start));
156 out.push_back(t);
157 continue;
158 }
159 ++i;
160 }
161 out.push_back({CalcToken::Kind::End});
162 return out;
163 }

186 class CalcParser
187 {
188 public:
189 explicit CalcParser(const std::vector<CalcToken> &toks) : toks(toks) {}
190
191 bool parse(Value &out)
192 {
193 out = parseExpr();
194 return !errored && pos < toks.size() && toks[pos].kind == CalcToken::Kind::End;
195 }
196
197 private:
198 const std::vector<CalcToken> &toks;
199 size_t pos{0};
200 bool errored{false};
201
202 const CalcToken &peek() const { return toks[pos]; }
203 void advance()
204 {
205 if (pos + 1 < toks.size())
206 ++pos;
207 }
208 void error() { errored = true; }
209
210 Value parseExpr()
211 {
212 Value left = parseTerm();
213 while (!errored && (peek().kind == CalcToken::Kind::Plus ||
214 peek().kind == CalcToken::Kind::Minus))
215 {
216 auto op = peek().kind;
217 advance();
218 Value right = parseTerm();
219 left = (op == CalcToken::Kind::Plus) ? (left + right) : (left - right);
220 }
221 return left;
222 }
223
224 Value parseTerm()
225 {
226 Value left = parseFactor();
227 while (!errored && (peek().kind == CalcToken::Kind::Star ||
228 peek().kind == CalcToken::Kind::Slash))
229 {
230 auto op = peek().kind;
231 advance();
232 Value right = parseFactor();
233 left = (op == CalcToken::Kind::Star) ? left.mulWith(right)
234 : left.divWith(right);
235 }
236 return left;
237 }
238
239 Value parseFactor()
240 {
241 if (errored)
242 return Value::px(0.0f);
243 if (peek().kind == CalcToken::Kind::Minus)
244 {
245 advance();
246 return -parseFactor();
247 }
248 if (peek().kind == CalcToken::Kind::Plus)
249 {
250 advance();
251 return parseFactor();
252 }
253 if (peek().kind == CalcToken::Kind::LParen)
254 {
255 advance();
256 Value v = parseExpr();
257 if (peek().kind != CalcToken::Kind::RParen)
258 {
259 error();
260 return v;
261 }
262 advance();
263 return v;
264 }
265 if (peek().kind == CalcToken::Kind::Ident)
266 {
267 std::string func = peek().ident;
268 advance();
269 if (peek().kind == CalcToken::Kind::LParen)
270 {
271 advance();
272 std::vector<Value> fargs;
273 if (peek().kind != CalcToken::Kind::RParen)
274 {
275 fargs.push_back(parseExpr());
276 while (!errored && peek().kind == CalcToken::Kind::Comma)
277 {
278 advance();
279 fargs.push_back(parseExpr());
280 }
281 }
282 if (peek().kind != CalcToken::Kind::RParen)
283 {
284 error();
285 return Value::px(0.0f);
286 }
287 advance();
288
289 Value r;
290 r.args = std::make_shared<std::vector<Value>>(std::move(fargs));
291 if (func == "calc")
292 {
293 if (r.args->size() == 1)
294 return r.args->at(0);
295 else
296 {
297 error();
298 return Value::px(0.0f);
299 }
300 }
301 else if (func == "min")
302 r.op = ValueOp::Min;
303 else if (func == "max")
304 r.op = ValueOp::Max;
305 else if (func == "clamp")
306 r.op = ValueOp::Clamp;
307 else if (func == "round")
308 r.op = ValueOp::Round;
309 else if (func == "floor")
310 r.op = ValueOp::Floor;
311 else if (func == "ceil")
312 r.op = ValueOp::Ceil;
313 else if (func == "mod")
314 r.op = ValueOp::Mod;
315 else
316 {
317 error();
318 return Value::px(0.0f);
319 }
320 return r;
321 }
322 error();
323 return Value::px(0.0f);
324 }
325 if (peek().kind == CalcToken::Kind::Number)
326 {
327 float num = peek().num;
328 advance();
329 if (peek().kind == CalcToken::Kind::Percent)
330 {
331 advance();
332 return Value::percent(num);
333 }
334 if (peek().kind == CalcToken::Kind::Ident)
335 {
336 const std::string &u = peek().ident;
337 advance();
338 if (u == "px")
339 return Value::px(num);
340 if (u == "vw")
341 return Value::vw(num);
342 if (u == "vh")
343 return Value::vh(num);
344 if (u == "pw")
345 return Value::pw(num);
346 if (u == "ph")
347 return Value::ph(num);
348 return Value::px(num);
349 }
350 return Value::number(num);
351 }
352 error();
353 return Value::px(0.0f);
354 }

360 inline FontStack parseFontStack(std::string_view raw)
361 {
362 FontStack out;
363 size_t i = 0;
364 while (i < raw.size())
365 {
366 while (i < raw.size() &&
367 (std::isspace((unsigned char)raw[i]) || raw[i] == ','))
368 ++i;
369 if (i >= raw.size())
370 break;
371
372 char quote = 0;
373 if (raw[i] == '"' || raw[i] == '\'')
374 {
375 quote = raw[i];
376 ++i;
377 }
378
379 size_t start = i;
380 if (quote)
381 {
382 while (i < raw.size() && raw[i] != quote)
383 ++i;
384 out.push_back(std::string(raw.substr(start, i - start)));
385 if (i < raw.size())
386 ++i;
387 }
388 else
389 {
390 while (i < raw.size() &&
391 raw[i] != ',' &&
392 !std::isspace((unsigned char)raw[i]))
393 ++i;
394 out.push_back(std::string(raw.substr(start, i - start)));
395 }
396 }
397 return out;
398 }

406 inline std::optional<Value> parseValueToken(std::string_view s)
407 {
408 if (s.empty())
409 return std::nullopt;
410 if (s == "auto")
411 return Value::autoSize();
412
413 if (s.starts_with("calc(") || s.starts_with("min(") || s.starts_with("max(") ||
414 s.starts_with("clamp(") || s.starts_with("round(") || s.starts_with("floor(") ||
415 s.starts_with("ceil(") || s.starts_with("mod("))
416 {
417
418 auto toks = detail::tokenizeCalc(s);
419 detail::CalcParser p(toks);
420 Value out;
421 if (!p.parse(out))
422 {
423 logWarn("StyleParser", "", 0, 0, "invalid math expression '" + std::string(s) + "'");
424 return std::nullopt;
425 }
426 return out;
427 }
428
429 size_t n = 0;
430 while (n < s.size() && (std::isdigit((unsigned char)s[n]) ||
431 s[n] == '.' || s[n] == '-' || s[n] == '+'))
432 ++n;
433
434 float amount = 0.0f;
435 if (!detail::parseFloatSV(s.substr(0, n), amount))
436 return std::nullopt;
437
438 std::string_view unit = s.substr(n);
439 if (unit.empty() || unit == "px")
440 return Px(amount);
441 if (unit == "%")
442 return Percent(amount);
443 if (unit == "vw")
444 return VW(amount);
445 if (unit == "vh")
446 return VH(amount);
447 if (unit == "pw")
448 return PW(amount);
449 if (unit == "ph")
450 return PH(amount);
451 return std::nullopt;
452 }

458 inline std::optional<Color> parseColorToken(std::string_view s)
459 {
460 if (s == "white")
461 return Colors::White;
462 if (s == "black")
463 return Colors::Black;
464 if (s == "transparent")
465 return Colors::Blank;
466 if (s == "blank")
467 return Colors::Blank;
468 if (s == "red")
469 return Colors::Red;
470 if (s == "maroon")
471 return Colors::Maroon;
472 if (s == "green")
473 return Colors::Green;
474 if (s == "blue")
475 return Colors::Blue;
476 if (s == "yellow")
477 return Colors::Yellow;
478 if (s == "gray")
479 return Colors::Gray;
480 if (s == "darkgray")
481 return Colors::DarkGray;
482 if (s == "lightgray")
483 return Colors::LightGray;
484 if (s.size() > 4 && s[0] == 'r' && s[1] == 'g' && s[2] == 'b')
485 {
486 bool hasAlpha = (s[3] == 'a');
487 size_t paren = s.find('(');
488 if (paren != std::string_view::npos && s.back() == ')')
489 {
490 std::string_view inner = s.substr(paren + 1, s.size() - paren - 2);
491 std::vector<float> parts;
492 std::string cur;
493 for (char c : inner)
494 {
495 if (c == ',')
496 {
497 try
498 {
499 parts.push_back(std::stof(cur));
500 }
501 catch (...)
502 {
503 return std::nullopt;
504 }
505 cur.clear();
506 }
507 else if (!std::isspace((unsigned char)c))
508 {
509 cur.push_back(c);
510 }
511 }
512 if (!cur.empty())
513 {
514 try
515 {
516 parts.push_back(std::stof(cur));
517 }
518 catch (...)
519 {
520 return std::nullopt;
521 }
522 }
523
524 const size_t expected = hasAlpha ? 4 : 3;
525 if (parts.size() != expected)
526 return std::nullopt;
527
528 auto clamp255 = [](float v) -> uint8_t
529 {
530 return (uint8_t)std::clamp((int)std::lround(v), 0, 255);
531 };
532
533 uint8_t r = clamp255(parts[0]);
534 uint8_t g = clamp255(parts[1]);
535 uint8_t b = clamp255(parts[2]);
536 uint8_t a = hasAlpha
537 ? clamp255(parts[3] * 255.0f)
538 : (uint8_t)255;
539
540 return Color{r, g, b, a};
541 }
542 }
543 if (!s.empty() && s[0] == '#')
544 {
545 std::string h(s.substr(1));
546 auto hex = [](char c) -> int
547 {
548 if (c >= '0' && c <= '9')
549 return c - '0';
550 if (c >= 'a' && c <= 'f')
551 return c - 'a' + 10;
552 if (c >= 'A' && c <= 'F')
553 return c - 'A' + 10;
554 return 0;
555 };
556 if (h.size() == 3)
557 return Color{(uint8_t)(hex(h[0]) * 17), (uint8_t)(hex(h[1]) * 17), (uint8_t)(hex(h[2]) * 17), 255};
558 if (h.size() == 4)
559 return Color{(uint8_t)(hex(h[0]) * 17), (uint8_t)(hex(h[1]) * 17), (uint8_t)(hex(h[2]) * 17), (uint8_t)(hex(h[3]) * 17)};
560 if (h.size() == 6)
561 return Color{(uint8_t)(hex(h[0]) * 16 + hex(h[1])), (uint8_t)(hex(h[2]) * 16 + hex(h[3])), (uint8_t)(hex(h[4]) * 16 + hex(h[5])), 255};
562 if (h.size() == 8)
563 return Color{(uint8_t)(hex(h[0]) * 16 + hex(h[1])), (uint8_t)(hex(h[2]) * 16 + hex(h[3])), (uint8_t)(hex(h[4]) * 16 + hex(h[5])), (uint8_t)(hex(h[6]) * 16 + hex(h[7]))};
564 }
565 return std::nullopt;
566 }

572 inline std::optional<Spacing> parseSpacingToken(std::string_view s)
573 {
574 std::vector<Value> vals;
575 size_t i = 0;
576 while (i < s.size())
577 {
578 while (i < s.size() && s[i] == ' ')
579 ++i;
580 size_t start = i;
581 while (i < s.size() && s[i] != ' ')
582 ++i;
583 if (start == i)
584 break;
585 auto v = parseValueToken(s.substr(start, i - start));
586 if (!v)
587 return std::nullopt;
588 vals.push_back(*v);
589 }
590 if (vals.size() == 1)
591 return Spacing(vals[0]);
592 if (vals.size() == 2)
593 return Spacing(vals[0], vals[1]);
594 if (vals.size() == 3)
595 return Spacing(vals[0], vals[1], vals[2]);
596 if (vals.size() == 4)
597 return Spacing(vals[0], vals[1], vals[2], vals[3]);
598 return std::nullopt;
599 }

1056 inline std::string substituteVarRefs(
1057 std::string v,
1058 const std::unordered_map<std::string, std::string> &customProps)
1059 {
1060 auto trimLocal = [](std::string s)
1061 {
1062 size_t a = 0, b = s.size();
1063 while (a < b && std::isspace((unsigned char)s[a]))
1064 a++;
1065 while (b > a && std::isspace((unsigned char)s[b - 1]))
1066 b--;
1067 return s.substr(a, b - a);
1068 };
1069
1070 bool changed = true;
1071 int safety = 0;
1072 while (changed && safety++ < 10)
1073 {
1074 changed = false;
1075 size_t pos = 0;
1076 while ((pos = v.find("var(", pos)) != std::string::npos)
1077 {
1078 size_t close = v.find(')', pos);
1079 if (close == std::string::npos)
1080 break;
1081 std::string name = trimLocal(v.substr(pos + 4, close - pos - 4));
1082 auto it = customProps.find(name);
1083 if (it == customProps.end())
1084 {
1085 pos = close + 1;
1086 continue;
1087 }
1088 v.replace(pos, close - pos + 1, it->second);
1089 changed = true;
1090 }
1091 }
1092 return v;
1093 }
