1205 virtual void renderChrome(float op, const ComputedStyle &style)
1206 {
1207 auto r = UIContext::get().renderer;
1208 if (!r)
1209 return;
1210 if (rect.width <= 0 || rect.height <= 0)
1211 return;
1212
1213 float maxRadius = std::min(rect.width, rect.height) * 0.5f;
1214 float rPx = std::clamp(style.radius.resolveH(maxRadius * 2.0f, maxRadius * 2.0f), 0.0f, maxRadius);
1215
1216 if (style.boxShadow.enabled)
1217 {
1218 const auto &sh = style.boxShadow;
1219 float sx = sh.x.resolveSelfH(rect.width, rect.height);
1220 float sy = sh.y.resolveSelfV(rect.width, rect.height);
1221 float blur = sh.blur.resolveSelfH(rect.width, rect.height);
1222
1223 Rect shadowRect = {
1224 rect.x + sx,
1225 rect.y + sy,
1226 rect.width,
1227 rect.height};
1228
1229 Color baseColor = sh.color.withAlpha(op);
1230
1231 if (blur <= 0.5f)
1232 {
1233 r->fillRoundedRect(shadowRect, rPx, baseColor);
1234 }
1235 else
1236 {
1237 constexpr int LAYERS = 5;
1238 for (int i = 0; i < LAYERS; ++i)
1239 {
1240 float expand = blur * (float)(i + 1) / (float)LAYERS;
1241 float alpha = (1.0f - (float)i / (float)LAYERS) * 0.35f;
1242 Rect r2 = {
1243 shadowRect.x - expand,
1244 shadowRect.y - expand,
1245 shadowRect.width + expand * 2.0f,
1246 shadowRect.height + expand * 2.0f};
1247 float rPx2 = std::clamp(rPx + expand, 0.0f, expand + maxRadius);
1248 r->fillRoundedRect(r2, rPx2, sh.color.withAlpha(op * alpha));
1249 }
1250 }
1251 }
1252
1253 Color bg = style.background.withAlpha(op);
1254 if (bg.a > 0)
1255 {
1256 if (rPx > 0.0f && maxRadius > 0.0f)
1257 r->fillRoundedRect(rect, rPx, bg);
1258 else
1259 r->fillRect(rect, bg);
1260 }

1517 inline bool nodeMatchesSimple(const Layout *node, const SimpleSelector &ss)
1518 {
1519 if (!node)
1520 return false;
1521
1522 if (ss.requireHover && !node->isHoveredState())
1523 return false;
1524 if (ss.requirePressed && !node->isPressedState())
1525 return false;
1526 if (ss.requireFocus && !node->isFocusedState())
1527 return false;
1528 if (ss.requireDisabled && !node->isDisabledState())
1529 return false;
1530 if (ss.requireChecked && !node->isCheckedState())
1531 return false;
1532
1533 auto matchesComponent = [node](SimpleSelector::Kind kind,
1534 const std::string &name) -> bool
1535 {
1536 if (kind == SimpleSelector::Kind::Tag)
1537 return node->getStyleTag() == name;
1538 if (kind == SimpleSelector::Kind::Id)
1539 return node->nodeId == name;
1540 for (const auto &c : node->getStyleClasses())
1541 if (c == name)
1542 return true;
1543 return false;
1544 };
1545
1546 if (!matchesComponent(ss.kind, ss.name))
1547 return false;
1548
1549 for (const auto &[kind, name] : ss.extras)
1550 {
1551 if (!matchesComponent(kind, name))
1552 return false;
1553 }
1554
1555 return true;
1556 }
1557
1572 inline bool ruleMatches(const ThemeRule &r, const Layout *node)
1573 {
1574 if (r.chain.empty() || !node)
1575 return false;
1576 if (r.media.has_value() && !evaluateMedia(*r.media, Metrics::viewport))
1577 return false;
1578 if (!nodeMatchesSimple(node, r.chain.back()))
1579 return false;
1580
1581 int i = (int)r.chain.size() - 2;
1582 const Layout *cur = node->getParent().get();
1583 while (i >= 0 && cur)
1584 {
1585 if (nodeMatchesSimple(cur, r.chain[i]))
1586 --i;
1587 cur = cur->getParent().get();
1588 }
1589 return i < 0;
1590 }
