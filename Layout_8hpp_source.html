427 bool moveChildBefore(const Layout *anchor, const Layout *child)
428 {
429 if (!child)
430 return false;
431 auto it = std::find_if(children.begin(), children.end(),
432 [child](const std::shared_ptr<Layout> &c)
433 { return c.get() == child; });
434 if (it == children.end())
435 return false;
436 if (anchor == child)
437 return true;
438
439 auto anchorIt = std::find_if(children.begin(), children.end(),
440 [anchor](const std::shared_ptr<Layout> &c)
441 { return c.get() == anchor; });
442 if (std::next(it) == anchorIt)
443 return true;
444
445 if (it < anchorIt)
446 std::rotate(it, std::next(it), anchorIt);
447 else
448 std::rotate(anchorIt, it, std::next(it));
449
450 internal_.pendingTransition = 1;
451 markLayoutDirtyUpward();
452 return true;
453 }

1719 inline bool nodeMatchesSimple(const Layout *node, const SimpleSelector &ss)
1720 {
1721 if (!node)
1722 return false;
1723
1724 if (ss.requireHover && !node->isHoveredState())
1725 return false;
1726 if (ss.requirePressed && !node->isPressedState())
1727 return false;
1728 if (ss.requireFocus && !node->isFocusedState())
1729 return false;
1730 if (ss.requireDisabled && !node->isDisabledState())
1731 return false;
1732 if (ss.requireChecked && !node->isCheckedState())
1733 return false;
1734
1735 auto matchesComponent = [node](SimpleSelector::Kind kind,
1736 const std::string &name) -> bool
1737 {
1738 if (kind == SimpleSelector::Kind::Tag)
1739 return node->getStyleTag() == name;
1740 if (kind == SimpleSelector::Kind::Id)
1741 return node->nodeId == name;
1742 for (const auto &c : node->getStyleClasses())
1743 if (c == name)
1744 return true;
1745 return false;
1746 };
1747
1748 if (!matchesComponent(ss.kind, ss.name))
1749 return false;
1750
1751 for (const auto &[kind, name] : ss.extras)
1752 {
1753 if (!matchesComponent(kind, name))
1754 return false;
1755 }
1756
1757 return true;
1758 }

1774 inline bool ruleMatches(const ThemeRule &r, const Layout *node, Vec2 viewport)
1775 {
1776 if (r.chain.empty() || !node)
1777 return false;
1778 if (r.media.has_value() && !evaluateMedia(*r.media, viewport))
1779 return false;
1780 if (!nodeMatchesSimple(node, r.chain.back()))
1781 return false;
1782
1783 int i = (int)r.chain.size() - 2;
1784 const Layout *cur = node->getParent().get();
1785 while (i >= 0 && cur)
1786 {
1787 if (nodeMatchesSimple(cur, r.chain[i]))
1788 --i;
1789 cur = cur->getParent().get();
1790 }
1791 return i < 0;
1792 }
