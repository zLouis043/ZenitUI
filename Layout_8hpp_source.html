411 bool moveChildBefore(const Layout *anchor, const Layout *child)
412 {
413 if (!child)
414 return false;
415 auto it = std::find_if(children.begin(), children.end(),
416 [child](const std::shared_ptr<Layout> &c)
417 { return c.get() == child; });
418 if (it == children.end())
419 return false;
420 if (anchor == child)
421 return true;
422
423 auto anchorIt = std::find_if(children.begin(), children.end(),
424 [anchor](const std::shared_ptr<Layout> &c)
425 { return c.get() == anchor; });
426 if (std::next(it) == anchorIt)
427 return true;
428
429 if (it < anchorIt)
430 std::rotate(it, std::next(it), anchorIt);
431 else
432 std::rotate(anchorIt, it, std::next(it));
433
434 internal_.pendingTransition = 1;
435 markLayoutDirtyUpward();
436 return true;
437 }

1729 inline bool nodeMatchesSimple(const Layout *node, const SimpleSelector &ss)
1730 {
1731 if (!node)
1732 return false;
1733
1734 if (ss.requireHover && !node->isHoveredState())
1735 return false;
1736 if (ss.requirePressed && !node->isPressedState())
1737 return false;
1738 if (ss.requireFocus && !node->isFocusedState())
1739 return false;
1740 if (ss.requireDisabled && !node->isDisabledState())
1741 return false;
1742 if (ss.requireChecked && !node->isCheckedState())
1743 return false;
1744
1745 auto matchesComponent = [node](SimpleSelector::Kind kind,
1746 const std::string &name) -> bool
1747 {
1748 if (kind == SimpleSelector::Kind::Tag)
1749 return node->getStyleTag() == name;
1750 if (kind == SimpleSelector::Kind::Id)
1751 return node->nodeId == name;
1752 for (const auto &c : node->getStyleClasses())
1753 if (c == name)
1754 return true;
1755 return false;
1756 };
1757
1758 if (!matchesComponent(ss.kind, ss.name))
1759 return false;
1760
1761 for (const auto &[kind, name] : ss.extras)
1762 {
1763 if (!matchesComponent(kind, name))
1764 return false;
1765 }
1766
1767 return true;
1768 }

1784 inline bool ruleMatches(const ThemeRule &r, const Layout *node, Vec2 viewport)
1785 {
1786 if (r.chain.empty() || !node)
1787 return false;
1788 if (r.media.has_value() && !evaluateMedia(*r.media, viewport))
1789 return false;
1790 if (!nodeMatchesSimple(node, r.chain.back()))
1791 return false;
1792
1793 int i = (int)r.chain.size() - 2;
1794 const Layout *cur = node->getParent().get();
1795 while (i >= 0 && cur)
1796 {
1797 if (nodeMatchesSimple(cur, r.chain[i]))
1798 --i;
1799 cur = cur->getParent().get();
1800 }
1801 return i < 0;
1802 }
