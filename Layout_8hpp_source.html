1212 virtual void renderChrome(float op, const ComputedStyle &style)
1213 {

1215 if (!r)
1216 return;
1217 if (rect.width <= 0 || rect.height <= 0)
1218 return;
1219
1220 float maxRadius = std::min(rect.width, rect.height) * 0.5f;
1221 float rPx = std::clamp(style.radius.resolveH(maxRadius * 2.0f, maxRadius * 2.0f), 0.0f, maxRadius);
1222
1223 if (style.boxShadow.enabled)
1224 {
1225 const auto &sh = style.boxShadow;
1226 float sx = sh.x.resolveSelfH(rect.width, rect.height);
1227 float sy = sh.y.resolveSelfV(rect.width, rect.height);
1228 float blur = sh.blur.resolveSelfH(rect.width, rect.height);
1229
1230 Rect shadowRect = {
1231 rect.x + sx,
1232 rect.y + sy,
1233 rect.width,
1234 rect.height};
1235
1236 Color baseColor = sh.color.withAlpha(op);
1237
1238 if (blur <= 0.5f)
1239 {
1240 r->fillRoundedRect(shadowRect, rPx, baseColor);
1241 }
1242 else
1243 {
1244 constexpr int LAYERS = 5;
1245 for (int i = 0; i < LAYERS; ++i)
1246 {
1247 float expand = blur * (float)(i + 1) / (float)LAYERS;
1248 float alpha = (1.0f - (float)i / (float)LAYERS) * 0.35f;
1249 Rect r2 = {
1250 shadowRect.x - expand,
1251 shadowRect.y - expand,
1252 shadowRect.width + expand * 2.0f,
1253 shadowRect.height + expand * 2.0f};
1254 float rPx2 = std::clamp(rPx + expand, 0.0f, expand + maxRadius);
1255 r->fillRoundedRect(r2, rPx2, sh.color.withAlpha(op * alpha));
1256 }
1257 }
1258 }
1259
1260 Color bg = style.background.withAlpha(op);
1261 if (bg.a > 0)
1262 {
1263 if (rPx > 0.0f && maxRadius > 0.0f)
1264 r->fillRoundedRect(rect, rPx, bg);
1265 else
1266 r->fillRect(rect, bg);
1267 }

1533 inline bool nodeMatchesSimple(const Layout *node, const SimpleSelector &ss)
1534 {
1535 if (!node)
1536 return false;
1537
1538 if (ss.requireHover && !node->isHoveredState())
1539 return false;
1540 if (ss.requirePressed && !node->isPressedState())
1541 return false;
1542 if (ss.requireFocus && !node->isFocusedState())
1543 return false;
1544 if (ss.requireDisabled && !node->isDisabledState())
1545 return false;
1546 if (ss.requireChecked && !node->isCheckedState())
1547 return false;
1548
1549 auto matchesComponent = [node](SimpleSelector::Kind kind,
1550 const std::string &name) -> bool
1551 {
1552 if (kind == SimpleSelector::Kind::Tag)
1553 return node->getStyleTag() == name;
1554 if (kind == SimpleSelector::Kind::Id)
1555 return node->nodeId == name;
1556 for (const auto &c : node->getStyleClasses())
1557 if (c == name)
1558 return true;
1559 return false;
1560 };
1561
1562 if (!matchesComponent(ss.kind, ss.name))
1563 return false;
1564
1565 for (const auto &[kind, name] : ss.extras)
1566 {
1567 if (!matchesComponent(kind, name))
1568 return false;
1569 }
1570
1571 return true;
1572 }

1588 inline bool ruleMatches(const ThemeRule &r, const Layout *node)
1589 {
1590 if (r.chain.empty() || !node)
1591 return false;
1592 if (r.media.has_value() && !evaluateMedia(*r.media, Metrics::viewport))
1593 return false;
1594 if (!nodeMatchesSimple(node, r.chain.back()))
1595 return false;
1596
1597 int i = (int)r.chain.size() - 2;
1598 const Layout *cur = node->getParent().get();
1599 while (i >= 0 && cur)
1600 {
1601 if (nodeMatchesSimple(cur, r.chain[i]))
1602 --i;
1603 cur = cur->getParent().get();
1604 }
1605 return i < 0;
1606 }
