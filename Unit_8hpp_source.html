165 float resolveAST(float pw, float ph, float autoV, int mode) const
166 {
167 if (!args)
168 return 0.0f;
169 switch (op)
170 {
171 case ValueOp::Add:
172 return resolveArg(0, pw, ph, autoV, mode) + resolveArg(1, pw, ph, autoV, mode);
173 case ValueOp::Sub:
174 return resolveArg(0, pw, ph, autoV, mode) - resolveArg(1, pw, ph, autoV, mode);
175 case ValueOp::Mul:
176 return resolveArg(0, pw, ph, autoV, mode) * resolveArg(1, pw, ph, autoV, mode);
177 case ValueOp::Div:
178 {
179 float b = resolveArg(1, pw, ph, autoV, mode);
180 return b != 0.0f ? resolveArg(0, pw, ph, autoV, mode) / b : 0.0f;
181 }
182 case ValueOp::Min:
183 {
184 if (args->empty())
185 return 0.0f;
186 float r = resolveArg(0, pw, ph, autoV, mode);
187 for (size_t i = 1; i < args->size(); ++i)
188 r = std::min(r, resolveArg(i, pw, ph, autoV, mode));
189 return r;
190 }
191 case ValueOp::Max:
192 {
193 if (args->empty())
194 return 0.0f;
195 float r = resolveArg(0, pw, ph, autoV, mode);
196 for (size_t i = 1; i < args->size(); ++i)
197 r = std::max(r, resolveArg(i, pw, ph, autoV, mode));
198 return r;
199 }
200 case ValueOp::Clamp:
201 {
202 if (args->size() >= 3)
203 {
204 float minV = resolveArg(0, pw, ph, autoV, mode);
205 float valV = resolveArg(1, pw, ph, autoV, mode);
206 float maxV = resolveArg(2, pw, ph, autoV, mode);
207 return std::clamp(valV, minV, maxV);
208 }
209 return 0.0f;
210 }
211 case ValueOp::Round:
212 return args->empty() ? 0.0f : std::round(resolveArg(0, pw, ph, autoV, mode));
213 case ValueOp::Floor:
214 return args->empty() ? 0.0f : std::floor(resolveArg(0, pw, ph, autoV, mode));
215 case ValueOp::Ceil:
216 return args->empty() ? 0.0f : std::ceil(resolveArg(0, pw, ph, autoV, mode));
217 case ValueOp::Mod:
218 {
219 if (args->size() >= 2)
220 {
221 float a = resolveArg(0, pw, ph, autoV, mode);
222 float b = resolveArg(1, pw, ph, autoV, mode);
223 return b != 0.0f ? std::fmod(a, b) : 0.0f;
224 }
225 return 0.0f;
226 }
227 default:
228 return 0.0f;
229 }
230 }

548 void normalize()
549 {
550 float coeffs[7] = {0};
551 bool present[7] = {false};
552 for (const auto &t : terms)
553 {
554 int idx = static_cast<int>(t.unit);
555 if (idx < 0 || idx >= 7)
556 continue;
557 coeffs[idx] += t.coeff;
558 present[idx] = true;
559 }
560 terms.clear();
561 for (int i = 0; i < 7; ++i)
562 {
563 if (present[i] && coeffs[i] != 0.0f)
564 terms.push_back({coeffs[i], static_cast<Unit>(i)});
565 }
