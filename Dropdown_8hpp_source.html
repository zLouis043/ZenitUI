207 }
208 }
209 else if (k == Key::Up)
210 {
211 if (isOpen)
212 {

214 if (prev != selected)

216 setSelected(prev);
217 if (listContainer && prev < (int)listContainer->children.size())
218 listContainer->children[prev]->scrollIntoView();
219 }
220 }
221 }
222 else if (k == Key::Enter || k == Key::Space)
223 {
224 setOpen(!isOpen);
225 if (isOpen && listContainer && selected < (int)listContainer->children.size())
226 listContainer->children[selected]->scrollIntoView();

232 return;

234 if (button)

237 Layout *p = getParent().get();
238 while (p)
239 {
240 const auto &st = p->getStyle();
241 if (st.overflowX != Overflow::Visible || st.overflowY != Overflow::Visible)
242 {

244 if (br.y + br.height <= pr.y || br.y >= pr.y + pr.height ||
