754 void selectWordAtCursor()
755 {
756 if (text.empty())
757 return;
758 size_t s = Utf8::snapFloor(text, static_cast<size_t>(std::max(0, cursorPos)));
759 size_t e = s;
760 while (s > 0)
761 {
762 size_t prev = Utf8::prevCluster(text, s);
763 uint32_t cp = 0;
764 Utf8::decode(text, prev, cp);
765 if (!Utf8::isWordChar(cp))
766 break;
767 s = prev;
768 }
769 while (e < text.size())
770 {
771 uint32_t cp = 0;
772 Utf8::decode(text, e, cp);
773 if (!Utf8::isWordChar(cp))
774 break;
775 e = Utf8::nextCluster(text, e);
776 }
777 if (s != e)
778 {
779 selectionStart = static_cast<int>(s);
780 cursorPos = static_cast<int>(e);
781 }

784 static std::vector<std::string_view> wrapPlaceholderLines(const std::string &src, float availW,
785 FontHandle f, float fs, float sp)
786 {
787 std::vector<std::string_view> out;
788 auto r = UIContext::get().renderer;
789 if (!r || availW <= 1.0f || src.empty())
790 {
791 if (!src.empty())
792 out.emplace_back(src);
793 return out;
794 }
795
796 auto measure = [&](size_t a, size_t b) -> float
797 {
798 if (b <= a)
799 return 0.0f;
800 return r->measureText(f, std::string_view(src).substr(a, b - a), fs, sp).x;
801 };
802
803 size_t len = src.size();
804 size_t pStart = 0;
805 while (true)
806 {
807 size_t pEnd = pStart;
808 while (pEnd < len && src[pEnd] != '\n')
809 ++pEnd;
810
811 if (pStart == pEnd)
812 {
813 out.emplace_back();
814 }
815 else
816 {
817 auto bounds = Utf8::clusterBoundaries(std::string_view(src).substr(pStart, pEnd - pStart));
818 size_t lineStartIdx = 0;
819 const size_t lastIdx = bounds.size() - 1;
820 while (lineStartIdx < lastIdx)
821 {
822 size_t lineStart = pStart + bounds[lineStartIdx];
823 if (measure(lineStart, pEnd) <= availW)
824 {
825 out.emplace_back(std::string_view(src).substr(lineStart, pEnd - lineStart));
826 break;
827 }
828
829 size_t lo = lineStartIdx + 1;
830 size_t hi = lastIdx;
831 while (lo < hi)
832 {
833 size_t mid = lo + (hi - lo + 1) / 2;
834 if (measure(lineStart, pStart + bounds[mid]) <= availW)
835 lo = mid;
836 else
837 hi = mid - 1;
838 }
839 size_t fitIdx = std::max(lineStartIdx + 1, lo);
840 size_t breakIdx = fitIdx;
841
842 if (fitIdx < lastIdx && src[pStart + bounds[fitIdx]] == ' ')
843 {
844 breakIdx = fitIdx + 1;
845 }
846 else
847 {
848 for (size_t k = fitIdx - 1; k > lineStartIdx; --k)
849 {
850 if (src[pStart + bounds[k]] == ' ')
851 {
852 breakIdx = k + 1;
853 break;
854 }
855 }
856 }
857
858 size_t visEnd = pStart + bounds[breakIdx];
859 while (visEnd > lineStart && src[visEnd - 1] == ' ')
860 --visEnd;
861 out.emplace_back(std::string_view(src).substr(lineStart, visEnd - lineStart));
862 lineStartIdx = breakIdx;
863 }
864 }
865
866 if (pEnd >= len)
867 break;
868 pStart = pEnd + 1;
869 }
870 return out;
871 }

873 static std::string truncatePlaceholder(const std::string &src, float maxW,
874 FontHandle f, float fs, float sp)
875 {
876 auto r = UIContext::get().renderer;
877 if (!r || maxW <= 1.0f || src.empty())
878 return src;
879 if (r->measureText(f, src, fs, sp).x <= maxW)
880 return src;
881
882 const std::string ell = "...";
883 float ellW = r->measureText(f, ell, fs, sp).x;
884 if (ellW >= maxW)

886

888 size_t lo = 0, hi = bounds.size() - 1;
889 while (lo < hi)
890 {
891 size_t mid = lo + (hi - lo + 1) / 2;
892 if (r->measureText(f, std::string_view(src).substr(0, bounds[mid]), fs, sp).x + ellW <= maxW)
893 lo = mid;
894 else
895 hi = mid - 1;
896 }
897 size_t cut = bounds[lo];
898 while (cut > 0 && src[cut - 1] == ' ')
899 --cut;
900 return src.substr(0, cut) + ell;
901 }
