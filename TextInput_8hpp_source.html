747 void selectWordAtCursor()
748 {
749 if (text.empty())
750 return;
751 size_t s = Utf8::snapFloor(text, static_cast<size_t>(std::max(0, cursorPos)));
752 size_t e = s;
753 while (s > 0)
754 {
755 size_t prev = Utf8::prevCluster(text, s);
756 uint32_t cp = 0;
757 Utf8::decode(text, prev, cp);
758 if (!Utf8::isWordChar(cp))
759 break;
760 s = prev;
761 }
762 while (e < text.size())
763 {
764 uint32_t cp = 0;
765 Utf8::decode(text, e, cp);
766 if (!Utf8::isWordChar(cp))
767 break;
768 e = Utf8::nextCluster(text, e);
769 }
770 if (s != e)
771 {
772 selectionStart = static_cast<int>(s);
773 cursorPos = static_cast<int>(e);
774 }

777 static std::vector<std::string_view> wrapPlaceholderLines(const std::string &src, float availW,
778 FontHandle f, float fs, float sp)
779 {
780 std::vector<std::string_view> out;
781 auto r = UIContext::get().renderer;
782 if (!r || availW <= 1.0f || src.empty())
783 {
784 if (!src.empty())
785 out.emplace_back(src);
786 return out;
787 }
788
789 auto measure = [&](size_t a, size_t b) -> float
790 {
791 if (b <= a)
792 return 0.0f;
793 return r->measureText(f, std::string_view(src).substr(a, b - a), fs, sp).x;
794 };
795
796 size_t len = src.size();
797 size_t pStart = 0;
798 while (true)
799 {
800 size_t pEnd = pStart;
801 while (pEnd < len && src[pEnd] != '\n')
802 ++pEnd;
803
804 if (pStart == pEnd)
805 {
806 out.emplace_back();
807 }
808 else
809 {
810 auto bounds = Utf8::clusterBoundaries(std::string_view(src).substr(pStart, pEnd - pStart));
811 size_t lineStartIdx = 0;
812 const size_t lastIdx = bounds.size() - 1;
813 while (lineStartIdx < lastIdx)
814 {
815 size_t lineStart = pStart + bounds[lineStartIdx];
816 if (measure(lineStart, pEnd) <= availW)
817 {
818 out.emplace_back(std::string_view(src).substr(lineStart, pEnd - lineStart));
819 break;
820 }
821
822 size_t lo = lineStartIdx + 1;
823 size_t hi = lastIdx;
824 while (lo < hi)
825 {
826 size_t mid = lo + (hi - lo + 1) / 2;
827 if (measure(lineStart, pStart + bounds[mid]) <= availW)
828 lo = mid;
829 else
830 hi = mid - 1;
831 }
832 size_t fitIdx = std::max(lineStartIdx + 1, lo);
833 size_t breakIdx = fitIdx;
834
835 if (fitIdx < lastIdx && src[pStart + bounds[fitIdx]] == ' ')
836 {
837 breakIdx = fitIdx + 1;
838 }
839 else
840 {
841 for (size_t k = fitIdx - 1; k > lineStartIdx; --k)
842 {
843 if (src[pStart + bounds[k]] == ' ')
844 {
845 breakIdx = k + 1;
846 break;
847 }
848 }
849 }
850
851 size_t visEnd = pStart + bounds[breakIdx];
852 while (visEnd > lineStart && src[visEnd - 1] == ' ')
853 --visEnd;
854 out.emplace_back(std::string_view(src).substr(lineStart, visEnd - lineStart));
855 lineStartIdx = breakIdx;
856 }
857 }
858
859 if (pEnd >= len)
860 break;
861 pStart = pEnd + 1;
862 }
863 return out;
864 }

866 static std::string truncatePlaceholder(const std::string &src, float maxW,
867 FontHandle f, float fs, float sp)
868 {
869 auto r = UIContext::get().renderer;
870 if (!r || maxW <= 1.0f || src.empty())
871 return src;
872 if (r->measureText(f, src, fs, sp).x <= maxW)
873 return src;
874
875 const std::string ell = "...";
876 float ellW = r->measureText(f, ell, fs, sp).x;
877 if (ellW >= maxW)
