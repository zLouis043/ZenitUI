656 void selectWordAtCursor()
657 {
658 if (text.empty())
659 return;
660 size_t s = Utf8::snapFloor(text, static_cast<size_t>(std::max(0, cursorPos)));
661 size_t e = s;
662 while (s > 0)
663 {
664 size_t prev = Utf8::prevCluster(text, s);
665 uint32_t cp = 0;
666 Utf8::decode(text, prev, cp);
667 if (!Utf8::isWordChar(cp))
668 break;
669 s = prev;
670 }
671 while (e < text.size())
672 {
673 uint32_t cp = 0;
674 Utf8::decode(text, e, cp);
675 if (!Utf8::isWordChar(cp))
676 break;
677 e = Utf8::nextCluster(text, e);
678 }
679 if (s != e)
680 {
681 selectionStart = static_cast<int>(s);
682 cursorPos = static_cast<int>(e);
683 }
