59 inline Specificity computeSpecificity(const std::vector<SimpleSelector> &chain)
60 {
61 Specificity s;
62 for (const auto &sel : chain)
63 {
64 if (sel.kind == SimpleSelector::Kind::Id)
65 s.ids++;
66 else if (sel.kind == SimpleSelector::Kind::Class)
67 s.classes++;
68 else
69 s.tags++;
