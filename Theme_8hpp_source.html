55 inline Specificity computeSpecificity(const std::vector<SimpleSelector> &chain)
56 {
57 Specificity s;
58 for (const auto &sel : chain)
59 {
60 if (sel.kind == SimpleSelector::Kind::Id)
61 s.ids++;
62 else if (sel.kind == SimpleSelector::Kind::Class)
63 s.classes++;
64 else
65 s.tags++;
66
67 for (const auto &[kind, name] : sel.extras)
68 {
69 (void)name;

71 s.ids++;
72 else if (kind == SimpleSelector::Kind::Class)
73 s.classes++;
74 else

76 }

79 sel.requireDisabled || sel.requireChecked)
80 s.classes++;
81 }
