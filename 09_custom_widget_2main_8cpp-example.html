Rating.hpp

#pragma once
#include "Common.hpp"
#include "Layout.hpp"
namespace ZenitUI::UI {
// ---------------------------------------------------------------------------
// Rating — a custom widget that shows N stars and lets the user pick one.
//
// This file is a template for writing your own widget. It demonstrates:
//
// - TLayout<Derived> the CRTP factory: create() runs onBuild()
// - onBuild() one-time setup (style tag, flags, children)
// - computeIntrinsicSize() natural size when width/height are auto
// - renderContent() drawing the widget using ::part styles
// - onUpdate() per-frame logic: read the pointer, respond
// - ::part a named styling surface (see theme.zstyle)
//
// Build: make examples
// Run: make run-09_custom_widget
// ---------------------------------------------------------------------------
class Rating : public TLayout<Rating>
{
public:
Rating(int maxStars = 5, int value = 0)
: TLayout<Rating>(LayoutType::Horizontal),
maxStars_(maxStars),
value_(std::clamp(value, 0, maxStars))
{
setStyleTag("Rating");
setInteractive(true);
setBlocksRaycast(true);
setFocusable(true);
setKeyboardActivates(false); // we handle keys ourselves
getInlineBase().gap = Px(4.0f);
}
// ---- Public API ------------------------------------------------------
int value() const { return value_; }
int maxStars() const { return maxStars_; }
void setValue(int v)
{
int clamped = std::clamp(v, 0, maxStars_);
if (clamped == value_)
return;
value_ = clamped;
internal_.pendingTransition = 1;
if (onValueChanged)
onValueChanged(value_);
}
std::function<void(int)> onValueChanged;
protected:
// ---- Intrinsic size --------------------------------------------------
// Called during measure, before children are laid out. Returns the
// size the widget would like if it has no explicit width/height.
Vec2 computeIntrinsicSize(float, float) override
{
const Vec2 vp = viewportSize();
float fs = style_.currentStyle.fontSize.resolveV(
vp.x, vp.y, vp);
float starSize = fs * 1.4f;
float gap = getStyle().gap.resolveH(0, 0,vp);
float totalW = starSize * maxStars_ + gap * (maxStars_ - 1);
return { totalW, starSize };
}
// ---- Drawing ---------------------------------------------------------
// Called during draw, after the background (renderChrome) and before
// the children (there are none here).
void renderContent(float op, const ComputedStyle& style) override
{
auto r = UIContext::get().renderer;
if (!r) return;
const Vec2 vp = viewportSize();
float fs = style.fontSize.resolveV(vp.x, vp.y, vp);
float starSize = fs * 1.4f;
float gap = style.gap.resolveH(rect.width, rect.height, vp);
float totalW = starSize * maxStars_ + gap * (maxStars_ - 1);
float startX = rect.x + (rect.width - totalW) * 0.5f;
float centerY = rect.center().y;
// partStyle() resolves a ::part from the theme, with its own
// transitions and animations. Every part is a Style with optional
// fields: check is_set before using.
Style starStyle = partStyle("star");
Color filledColor = starStyle.color.is_set
? starStyle.color.value
: Color{255, 200, 60, 255};
Color emptyColor = starStyle.background.is_set
? starStyle.background.value
: Color{70, 70, 80, 255};
for (int i = 0; i < maxStars_; ++i)
{
float cx = startX + i * (starSize + gap) + starSize * 0.5f;
Color c = (i < value_) ? filledColor : emptyColor;
r->fillCircle({cx, centerY}, starSize * 0.5f, c.withAlpha(op));
}
}
// ---- Per-frame logic -------------------------------------------------
// Called after the state machine, every frame the node is enabled.
void onUpdate(float) override
{
auto& ctx = UIContext::get();
// Keyboard: Left / Right adjust, when focused.
if (isFocusedState())
{
for (int k : ctx.inputEvents.keys)
{
if (k == Key::Left) setValue(value_ - 1);
if (k == Key::Right) setValue(value_ + 1);
}
}
// Mouse: click picks the star under the pointer.
// Using topmostConsumer (set by hitTest) instead of a geometric
// rect.contains check ensures only the node that actually
// received the click reacts, even if some other node's rect
// overlaps the same point.
if (!ctx.pointer[MouseButton::Left].pressed) return;
if (ctx.topmostConsumer != this) return;
if (rect.width <= 0.0f) return;
const Vec2 vp = viewportSize();
float fs = style_.currentStyle.fontSize.resolveV(
vp.x, vp.y, vp);
float starSize = fs * 1.4f;
float gap = style_.currentStyle.gap.resolveH(rect.width, rect.height, vp);
float totalW = starSize * maxStars_ + gap * (maxStars_ - 1);
float startX = rect.x + (rect.width - totalW) * 0.5f;
float localX = ctx.pointer.pos.x - startX;
int idx = (int)(localX / (starSize + gap));
idx = std::clamp(idx, 0, maxStars_ - 1);
setValue(idx + 1);
}
private:
int maxStars_;
int value_;
};
} // namespace ZenitUI::UI

// @doc Custom Rating widget derived from TLayout<Derived>, demonstrating computeIntrinsicSize, renderContent with ::part, pointer/keyboard input, and ZMarkup registration.
// @group grp_core grp_widgets
// ZenitUI — Example 09: Custom widget
//
// Walks through writing a widget from scratch. The Rating widget lives in
// Rating.hpp next to this source, and demonstrates the full extension API:
// TLayout, onBuild, computeIntrinsicSize, renderContent with ::part,
// onUpdate with pointer and keyboard handling, and an onValueChanged
// callback.
//
// The example also shows how to use a custom widget from C++ and (at the
// bottom) how to register it as a ZMarkup tag.
//
// Build: make examples
// Run: make run-09_custom_widget
#include <raylib.h>
#include "RaylibBackend.hpp"
#include "ZenitUI.hpp"
#include "Rating.hpp"
#include <cstdio>
using namespace ZenitUI;
using namespace ZenitUI::UI;
// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------
static std::shared_ptr<Layout> Row(const std::string& label,
std::shared_ptr<Layout> content)
{
auto row = std::make_shared<Layout>(LayoutType::Horizontal);
row->cls("row");
auto l = Label(label);
l->cls("row-label");
row->addChild(l);
row->addChild(content);
return row;
}
// ---------------------------------------------------------------------------
// Main
// ---------------------------------------------------------------------------
int main()
{
InitWindow(800, 480, "ZenitUI — 09 Custom widget");
SetTargetFPS(60);
RaylibRenderer renderer;
RaylibPlatform platform;
RaylibAssetProvider assets(renderer);
UIContext::get().renderer = &renderer;
UIContext::get().platform = &platform;
UIContext::get().assets = &assets;
ZMarkup::loadStyleFile("examples/09_custom_widget/theme.zstyle", UIContext::get().theme);
// -----------------------------------------------------------------------
// Root + page
// -----------------------------------------------------------------------
auto root = std::make_shared<Layout>(LayoutType::Stack);
root->setStyleTag("Stack");
root->id("root");
auto page = std::make_shared<Layout>(LayoutType::Vertical);
page->cls("page");
auto panel = std::make_shared<Layout>(LayoutType::Vertical);
panel->cls("panel");
auto title = Label("Custom widget: Rating");
title->cls("title");
panel->addChild(title);
// -----------------------------------------------------------------------
// One Rating, five stars, initially set to 3.
// -----------------------------------------------------------------------
auto valueLabel = Label("3 / 5");
valueLabel->cls("value-label");
auto rating = Rating::create(5, 3);
rating->onValueChanged = [valueLabel](int v) {
valueLabel->setText(std::to_string(v) + " / 5");
};
auto content = std::make_shared<Layout>(LayoutType::Horizontal);
content->getInlineBase().gap = Px(16);
content->getInlineBase().itemsV = Align::Center;
content->addChild(rating);
content->addChild(valueLabel);
panel->addChild(Row("Click, or arrow keys", content));
// -----------------------------------------------------------------------
// A second Rating, ten stars, to show that the widget scales.
// -----------------------------------------------------------------------
{
auto r2 = Rating::create(10, 7);
r2->getInlineBase().fontSize = Px(20); // smaller stars
panel->addChild(Row("Ten stars", r2));
}
page->addChild(panel);
root->addChild(page);
// -----------------------------------------------------------------------
// ZMarkup registration
//
// The built-in ZMarkup::build uses a private registry, so to use a
// custom tag you build the tree manually with your own Registry.
// The block below is commented out to keep the example focused, but
// it shows the pattern.
// -----------------------------------------------------------------------
//
// ZMarkup::Registry reg;
// ZMarkup::registerBuiltins(reg);
// reg.reg("Rating", [](const ZMarkup::Element& e) {
// return Rating::create(
// (int)e.attrFloat("max", 5.0f),
// (int)e.attrFloat("value", 0.0f)
// );
// });
//
// auto el = ZMarkup::parse(R"(Rating#stars max=5 value=3)");
// auto ctx = std::make_shared<ZMarkup::BuildContext>();
// auto node = reg.create(el, *ctx);
// ZMarkup::UINode ui(node, ctx);
//
// ui.find<Rating>("stars")->onValueChanged = [](int v){ /* ... */ };
// panel->addChild(ui.root());
// -----------------------------------------------------------------------
// Main loop
// -----------------------------------------------------------------------
while (!WindowShouldClose())
{
root->runFrame(GetFrameTime());
BeginDrawing();
ClearBackground(::BLACK);
root->renderFrame();
EndDrawing();
}
CloseWindow();
return 0;
}
