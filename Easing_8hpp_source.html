45 inline float getRatio(float t, TransitionFunction fn)
46 {
47 if (t <= 0.0f)
48 return 0.0f;
49 if (t >= 1.0f)
50 return 1.0f;
51 constexpr float pi = 3.14159265358979323846f;
52 switch (fn)
53 {
54 case TransitionFunction::Linear:
55 return t;
56 case TransitionFunction::EaseInQuad:
57 return t * t;
58 case TransitionFunction::EaseOutQuad:
59 return t * (2.0f - t);
60 case TransitionFunction::EaseInOutQuad:
61 return t < 0.5f ? 2.0f * t * t : 1.0f - std::pow(-2.0f * t + 2.0f, 2.0f) / 2.0f;
62 case TransitionFunction::EaseInCubic:
63 return t * t * t;
64 case TransitionFunction::EaseOutCubic:
65 return 1.0f - std::pow(1.0f - t, 3.0f);
66 case TransitionFunction::EaseInOutCubic:
67 return t < 0.5f ? 4.0f * t * t * t : 1.0f - std::pow(-2.0f * t + 2.0f, 3.0f) / 2.0f;
68 case TransitionFunction::EaseInBack:
69 {
70 const float c1 = 1.70158f, c3 = c1 + 1.0f;
71 return c3 * t * t * t - c1 * t * t;
72 }
73 case TransitionFunction::EaseOutBack:
74 {
75 const float c1 = 1.70158f, c3 = c1 + 1.0f;
76 float f = t - 1.0f;
77 return 1.0f + c3 * f * f * f + c1 * f * f;
78 }
79 case TransitionFunction::EaseOutElastic:
80 {
81 const float c4 = (2.0f * pi) / 3.0f;
82 return std::pow(2.0f, -10.0f * t) * std::sin((t * 10.0f - 0.75f) * c4) + 1.0f;
83 }
84 case TransitionFunction::EaseOutBounce:
85 {
86 const float n1 = 7.5625f, d1 = 2.75f;
87 if (t < 1.0f / d1)
88 return n1 * t * t;
89 if (t < 2.0f / d1)
90 {
91 t -= 1.5f / d1;
92 return n1 * t * t + 0.75f;
93 }
94 if (t < 2.5f / d1)
95 {
96 t -= 2.25f / d1;
97 return n1 * t * t + 0.9375f;
98 }
99 t -= 2.625f / d1;
100 return n1 * t * t + 0.984375f;
101 }
102 }
103 return t;
