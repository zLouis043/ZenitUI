48 inline float getRatio(float t, TransitionFunction fn)
49 {
50 if (t <= 0.0f)
51 return 0.0f;
52 if (t >= 1.0f)
53 return 1.0f;
54 constexpr float pi = 3.14159265358979323846f;
55 switch (fn)
56 {
57 case TransitionFunction::Linear:
58 return t;
59 case TransitionFunction::EaseInQuad:
60 return t * t;
61 case TransitionFunction::EaseOutQuad:
62 return t * (2.0f - t);
63 case TransitionFunction::EaseInOutQuad:
64 return t < 0.5f ? 2.0f * t * t : 1.0f - std::pow(-2.0f * t + 2.0f, 2.0f) / 2.0f;
65 case TransitionFunction::EaseInCubic:
66 return t * t * t;
67 case TransitionFunction::EaseOutCubic:
68 return 1.0f - std::pow(1.0f - t, 3.0f);
69 case TransitionFunction::EaseInOutCubic:
70 return t < 0.5f ? 4.0f * t * t * t : 1.0f - std::pow(-2.0f * t + 2.0f, 3.0f) / 2.0f;
71 case TransitionFunction::EaseInBack:
72 {
73 const float c1 = 1.70158f, c3 = c1 + 1.0f;
74 return c3 * t * t * t - c1 * t * t;
75 }
76 case TransitionFunction::EaseOutBack:
77 {
78 const float c1 = 1.70158f, c3 = c1 + 1.0f;
79 float f = t - 1.0f;
80 return 1.0f + c3 * f * f * f + c1 * f * f;
81 }
82 case TransitionFunction::EaseOutElastic:
83 {
84 const float c4 = (2.0f * pi) / 3.0f;
85 return std::pow(2.0f, -10.0f * t) * std::sin((t * 10.0f - 0.75f) * c4) + 1.0f;
86 }
87 case TransitionFunction::EaseOutBounce:
88 {
89 const float n1 = 7.5625f, d1 = 2.75f;
90 if (t < 1.0f / d1)
91 return n1 * t * t;
92 if (t < 2.0f / d1)
93 {
94 t -= 1.5f / d1;
95 return n1 * t * t + 0.75f;
96 }
97 if (t < 2.5f / d1)
98 {
99 t -= 2.25f / d1;
100 return n1 * t * t + 0.9375f;
101 }
102 t -= 2.625f / d1;
103 return n1 * t * t + 0.984375f;
104 }
105 }
106 return t;
107 }
