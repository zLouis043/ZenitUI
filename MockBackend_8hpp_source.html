216 TextureHandle createTexture(int w, int h, const uint8_t *rgbaPixels = nullptr) override
217 {
218 if (w <= 0 || h <= 0)
219 return {};
220 const uint32_t id = nextTextureId_++;
221 DynamicTexture tex;
222 tex.width = w;
223 tex.height = h;
224 const size_t bytes = static_cast<size_t>(w) * static_cast<size_t>(h) * 4u;
225 if (rgbaPixels)
226 tex.pixels.assign(rgbaPixels, rgbaPixels + bytes);
227 else
228 tex.pixels.assign(bytes, 0u);
229 dynamicTextures[id] = std::move(tex);
230 ++createTextureCount;
231 ++liveTextures;
232 return TextureHandle{id, w, h};

234
235 void updateTexture(TextureHandle t, Rect region, const uint8_t *rgbaPixels) override
236 {
237 if (!t.valid() || !rgbaPixels)
238 return;
239 auto it = dynamicTextures.find(t.id);
240 if (it == dynamicTextures.end())
241 return;
242 auto &tex = it->second;
243 const int rx = static_cast<int>(region.x);
244 const int ry = static_cast<int>(region.y);
245 const int rw = static_cast<int>(region.width);
246 const int rh = static_cast<int>(region.height);
247 if (rx < 0 || ry < 0 || rw <= 0 || rh <= 0 ||
248 rx + rw > tex.width || ry + rh > tex.height)
249 return;
250 for (int row = 0; row < rh; ++row)
251 {
252 const uint8_t *src = rgbaPixels + static_cast<size_t>(row) * static_cast<size_t>(rw) * 4u;
253 uint8_t *dst = tex.pixels.data() +
254 (static_cast<size_t>(ry + row) * static_cast<size_t>(tex.width) + static_cast<size_t>(rx)) * 4u;
255 std::copy_n(src, static_cast<size_t>(rw) * 4u, dst);
256 }
257 ++updateTextureCount;
258 }
