533 ctx.topmostConsumer = nullptr;
534 ctx.pressTarget = nullptr;

537 ctx.wheelConsumedThisFrame = false;
538 ctx.pointer = {};
539 ctx.inputEvents = {};
540 ctx.dt = 0.0f;
541 ctx.time = 0.0;
542 ctx.activePortals.clear();
543 ctx.framePortals.clear();
544 ctx.targetPool.clear(renderer, &ctx.renderStats);
545 ctx.renderStats.resetAll();
546

548

554 void frame(std::shared_ptr<Layout> root, float dt = 1.0f / 60.0f)
555 {
556 auto &ctx = UIContext::get();
557 ctx.beginFrame(dt);
558 root->updateTree(dt);
559 root->measure(platform.viewport.x, platform.viewport.y);
560 root->arrange({0.0f, 0.0f, platform.viewport.x, platform.viewport.y});
561 root->draw();
562 ctx.endFrame();
