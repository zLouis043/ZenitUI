296 inline Vec2 applyTransform(const Transform2D &t, Vec2 p)
297 {
298 float dx = p.x - t.pivot.x;
299 float dy = p.y - t.pivot.y;
300 dx *= t.scale;
301 dy *= t.scale;
302
303 float rad = t.rotationDeg * 3.14159265358979323846f / 180.0f;
304 float c = std::cos(rad), s = std::sin(rad);
305 float rx = dx * c - dy * s;
306 float ry = dx * s + dy * c;
307
308 return {rx + t.pivot.x + t.translate.x, ry + t.pivot.y + t.translate.y};
309 }
310
319 inline Vec2 applyInverseTransform(const Transform2D &t, Vec2 p)
320 {
321 p.x -= t.translate.x;
322 p.y -= t.translate.y;
323 float dx = p.x - t.pivot.x, dy = p.y - t.pivot.y;
324 float rad = -t.rotationDeg * 3.14159265358979323846f / 180.0f;
325 float c = std::cos(rad), s = std::sin(rad);
326 float rx = dx * c - dy * s;
327 float ry = dx * s + dy * c;
328 rx /= t.scale;
329 ry /= t.scale;
330 return {rx + t.pivot.x, ry + t.pivot.y};
331 }
