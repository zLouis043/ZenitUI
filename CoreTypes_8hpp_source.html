310 inline Vec2 applyTransform(const Transform2D &t, Vec2 p)
311 {
312 float dx = p.x - t.pivot.x;
313 float dy = p.y - t.pivot.y;
314 dx *= t.scale;
315 dy *= t.scale;
316
317 float rad = t.rotationDeg * 3.14159265358979323846f / 180.0f;
318 float c = std::cos(rad), s = std::sin(rad);
319 float rx = dx * c - dy * s;
320 float ry = dx * s + dy * c;
321
322 return {rx + t.pivot.x + t.translate.x, ry + t.pivot.y + t.translate.y};
323 }
324
333 inline Vec2 applyInverseTransform(const Transform2D &t, Vec2 p)
334 {
335 p.x -= t.translate.x;
336 p.y -= t.translate.y;
337 float dx = p.x - t.pivot.x, dy = p.y - t.pivot.y;
338 float rad = -t.rotationDeg * 3.14159265358979323846f / 180.0f;
339 float c = std::cos(rad), s = std::sin(rad);
340 float rx = dx * c - dy * s;
341 float ry = dx * s + dy * c;
342 rx /= t.scale;
343 ry /= t.scale;
344 return {rx + t.pivot.x, ry + t.pivot.y};
345 }
