57 inline size_t encode(uint32_t cp, std::string &out)
58 {
59 if (!isValidCodepoint(cp))
60 return 0;
61 if (cp <= 0x7Fu)
62 {
63 out.push_back(static_cast<char>(cp));
64 return 1;
65 }
66 if (cp <= 0x7FFu)
67 {
68 out.push_back(static_cast<char>(0xC0u | (cp >> 6)));
69 out.push_back(static_cast<char>(0x80u | (cp & 0x3Fu)));
70 return 2;
71 }
72 if (cp <= 0xFFFFu)
73 {
74 out.push_back(static_cast<char>(0xE0u | (cp >> 12)));
75 out.push_back(static_cast<char>(0x80u | ((cp >> 6) & 0x3Fu)));
76 out.push_back(static_cast<char>(0x80u | (cp & 0x3Fu)));
77 return 3;
78 }

81 out.push_back(static_cast<char>(0x80u | ((cp >> 6) & 0x3Fu)));

105 inline size_t decode(std::string_view s, size_t pos, uint32_t &outCp) noexcept
106 {
107 if (pos >= s.size())
108 {
109 outCp = 0;
110 return s.size();
111 }
112
113 const unsigned char b0 = static_cast<unsigned char>(s[pos]);
114 if (b0 < 0x80u)
115 {
116 outCp = b0;
117 return pos + 1;
118 }
119
120 if ((b0 & 0xE0u) == 0xC0u && pos + 1 < s.size())
121 {
122 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
123 if (isContinuation(b1))
124 {
125 uint32_t cp = ((b0 & 0x1Fu) << 6) | (b1 & 0x3Fu);
126 if (cp >= 0x80u)
127 {
128 outCp = cp;
129 return pos + 2;
130 }
131 }
132 }
133 else if ((b0 & 0xF0u) == 0xE0u && pos + 2 < s.size())
134 {
135 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
136 const unsigned char b2 = static_cast<unsigned char>(s[pos + 2]);
137 if (isContinuation(b1) && isContinuation(b2))
138 {
139 uint32_t cp = ((b0 & 0x0Fu) << 12) | ((b1 & 0x3Fu) << 6) | (b2 & 0x3Fu);
140 if (cp >= 0x800u && isValidCodepoint(cp))
141 {
142 outCp = cp;
143 return pos + 3;
144 }
145 }
146 }
147 else if ((b0 & 0xF8u) == 0xF0u && pos + 3 < s.size())
148 {
149 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
150 const unsigned char b2 = static_cast<unsigned char>(s[pos + 2]);
151 const unsigned char b3 = static_cast<unsigned char>(s[pos + 3]);
152 if (isContinuation(b1) && isContinuation(b2) && isContinuation(b3))
153 {
154 uint32_t cp = ((b0 & 0x07u) << 18) | ((b1 & 0x3Fu) << 12) |
155 ((b2 & 0x3Fu) << 6) | (b3 & 0x3Fu);
156 if (cp >= 0x10000u && isValidCodepoint(cp))
157 {
158 outCp = cp;
159 return pos + 4;
160 }
161 }
162 }
163
164 outCp = 0xFFFDu;
165 return pos + 1;
166 }

254 inline size_t nextCluster(std::string_view s, size_t pos) noexcept
255 {
256 if (pos >= s.size())
257 return s.size();
258
259 pos = snapFloor(s, pos);
260 uint32_t firstCp = 0;
261 size_t cur = decode(s, pos, firstCp);
262
263 // CRLF counts as a single cluster.
264 if (firstCp == '\r' && cur < s.size() && s[cur] == '\n')
265 return cur + 1;
266
267 // Regional Indicator pair (e.g. flag emojis).
268 if (isRegionalIndicator(firstCp) && cur < s.size())
269 {
270 uint32_t nextCp = 0;
271 size_t nextPos = decode(s, cur, nextCp);
272 if (isRegionalIndicator(nextCp))
273 cur = nextPos;
274 }
275
276 // Consume trailing combining marks, variation selectors, modifiers, and ZWJ chains.
277 while (cur < s.size())
278 {
279 uint32_t cp = 0;
280 size_t after = decode(s, cur, cp);
281 if (isClusterExtend(cp))
282 {
283 cur = after;
284 continue;
285 }
286 if (cp == 0x200Du && after < s.size()) // Zero-Width Joiner
287 {
288 uint32_t joinedCp = 0;
289 size_t afterJoined = decode(s, after, joinedCp);
290 cur = afterJoined;
291 continue;
292 }
293 break;
294 }
295 return cur;
296 }

304 inline size_t prevCluster(std::string_view s, size_t pos) noexcept
305 {
306 if (pos == 0)
307 return 0;
308 if (pos > s.size())
309 pos = s.size();
310
311 // If pos was mid-codepoint, snapping down already reaches a valid start < pos.
312 if (pos < s.size() && isContinuation(static_cast<unsigned char>(s[pos])))
313 return snapFloor(s, pos);
314
315 size_t cur = prevCodepoint(s, pos);
316 uint32_t cp = 0;
317 decode(s, cur, cp);
318
319 // CRLF as a single cluster.
320 if (cp == '\n' && cur > 0 && s[cur - 1] == '\r')
321 return cur - 1;
322
323 // Step back over any trailing extenders or ZWJ sequences.
324 while (cur > 0)
325 {
326 if (isClusterExtend(cp))
327 {
328 cur = prevCodepoint(s, cur);
329 decode(s, cur, cp);
330 continue;
331 }
332
333 // If immediately preceded by a ZWJ, step back over ZWJ + previous base.
334 size_t before = prevCodepoint(s, cur);
335 uint32_t beforeCp = 0;
336 decode(s, before, beforeCp);
337 if (beforeCp == 0x200Du && before > 0)
338 {
339 cur = prevCodepoint(s, before);
340 decode(s, cur, cp);
341 continue;
342 }
343 break;
344 }
345
346 // Regional Indicator pair: pair from the start of a consecutive RI run.
347 if (isRegionalIndicator(cp) && cur > 0)
348 {
349 size_t runCount = 0;
350 size_t scan = cur;
351 while (scan > 0)
352 {
353 size_t p = prevCodepoint(s, scan);
354 uint32_t pcp = 0;
355 decode(s, p, pcp);
356 if (!isRegionalIndicator(pcp))
357 break;
358 ++runCount;
359 scan = p;
360 }
361 if ((runCount % 2u) == 1u)
362 cur = prevCodepoint(s, cur);
363 }
364
365 return cur;
366 }

410 inline constexpr bool isWordChar(uint32_t cp) noexcept
411 {
412 if ((cp >= 'a' && cp <= 'z') || (cp >= 'A' && cp <= 'Z') ||
413 (cp >= '0' && cp <= '9') || cp == '_')
414 return true;
415 if (cp >= 0x00C0u && cp <= 0x00FFu)
416 return cp != 0x00D7u && cp != 0x00F7u; // exclude × and ÷
417 if (cp >= 0x0100u && cp <= 0x024Fu)
418 return true; // Latin Extended-A/B
419 if (isClusterExtend(cp))
420 return true;
421 if ((cp >= 0x0370u && cp <= 0x052Fu) || // Greek, Cyrillic
422 (cp >= 0x3040u && cp <= 0x30FFu) || // Hiragana, Katakana
423 (cp >= 0x4E00u && cp <= 0x9FFFu) || // CJK Unified Ideographs
424 (cp >= 0xAC00u && cp <= 0xD7AFu)) // Hangul Syllables
425 return true;
426 return false;
427 }
