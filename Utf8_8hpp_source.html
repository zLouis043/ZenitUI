60 inline size_t encode(uint32_t cp, std::string &out)
61 {
62 if (!isValidCodepoint(cp))
63 return 0;
64 if (cp <= 0x7Fu)
65 {
66 out.push_back(static_cast<char>(cp));
67 return 1;
68 }
69 if (cp <= 0x7FFu)
70 {
71 out.push_back(static_cast<char>(0xC0u | (cp >> 6)));
72 out.push_back(static_cast<char>(0x80u | (cp & 0x3Fu)));
73 return 2;
74 }
75 if (cp <= 0xFFFFu)
76 {
77 out.push_back(static_cast<char>(0xE0u | (cp >> 12)));
78 out.push_back(static_cast<char>(0x80u | ((cp >> 6) & 0x3Fu)));
79 out.push_back(static_cast<char>(0x80u | (cp & 0x3Fu)));
80 return 3;
81 }
82 out.push_back(static_cast<char>(0xF0u | (cp >> 18)));
83 out.push_back(static_cast<char>(0x80u | ((cp >> 12) & 0x3Fu)));
84 out.push_back(static_cast<char>(0x80u | ((cp >> 6) & 0x3Fu)));
85 out.push_back(static_cast<char>(0x80u | (cp & 0x3Fu)));

108 inline size_t decode(std::string_view s, size_t pos, uint32_t &outCp) noexcept
109 {
110 if (pos >= s.size())
111 {
112 outCp = 0;
113 return s.size();
114 }
115
116 const unsigned char b0 = static_cast<unsigned char>(s[pos]);
117 if (b0 < 0x80u)
118 {
119 outCp = b0;
120 return pos + 1;
121 }
122
123 if ((b0 & 0xE0u) == 0xC0u && pos + 1 < s.size())
124 {
125 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
126 if (isContinuation(b1))
127 {
128 uint32_t cp = ((b0 & 0x1Fu) << 6) | (b1 & 0x3Fu);
129 if (cp >= 0x80u)
130 {
131 outCp = cp;
132 return pos + 2;
133 }
134 }
135 }
136 else if ((b0 & 0xF0u) == 0xE0u && pos + 2 < s.size())
137 {
138 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
139 const unsigned char b2 = static_cast<unsigned char>(s[pos + 2]);
140 if (isContinuation(b1) && isContinuation(b2))
141 {
142 uint32_t cp = ((b0 & 0x0Fu) << 12) | ((b1 & 0x3Fu) << 6) | (b2 & 0x3Fu);
143 if (cp >= 0x800u && isValidCodepoint(cp))
144 {
145 outCp = cp;
146 return pos + 3;
147 }
148 }
149 }
150 else if ((b0 & 0xF8u) == 0xF0u && pos + 3 < s.size())
151 {
152 const unsigned char b1 = static_cast<unsigned char>(s[pos + 1]);
153 const unsigned char b2 = static_cast<unsigned char>(s[pos + 2]);
154 const unsigned char b3 = static_cast<unsigned char>(s[pos + 3]);
155 if (isContinuation(b1) && isContinuation(b2) && isContinuation(b3))
156 {
157 uint32_t cp = ((b0 & 0x07u) << 18) | ((b1 & 0x3Fu) << 12) |
158 ((b2 & 0x3Fu) << 6) | (b3 & 0x3Fu);
159 if (cp >= 0x10000u && isValidCodepoint(cp))
160 {
161 outCp = cp;
162 return pos + 4;
163 }
164 }
165 }
166
167 outCp = 0xFFFDu;
168 return pos + 1;
169 }

257 inline size_t nextCluster(std::string_view s, size_t pos) noexcept
258 {
259 if (pos >= s.size())
260 return s.size();
261
262 pos = snapFloor(s, pos);
263 uint32_t firstCp = 0;
264 size_t cur = decode(s, pos, firstCp);
265
266 // CRLF counts as a single cluster.
267 if (firstCp == '\r' && cur < s.size() && s[cur] == '\n')
268 return cur + 1;
269
270 // Regional Indicator pair (e.g. flag emojis).
271 if (isRegionalIndicator(firstCp) && cur < s.size())
272 {
273 uint32_t nextCp = 0;
274 size_t nextPos = decode(s, cur, nextCp);
275 if (isRegionalIndicator(nextCp))
276 cur = nextPos;
277 }
278
279 // Consume trailing combining marks, variation selectors, modifiers, and ZWJ chains.
280 while (cur < s.size())
281 {
282 uint32_t cp = 0;
283 size_t after = decode(s, cur, cp);
284 if (isClusterExtend(cp))
285 {
286 cur = after;
287 continue;
288 }
289 if (cp == 0x200Du && after < s.size()) // Zero-Width Joiner
290 {
291 uint32_t joinedCp = 0;
292 size_t afterJoined = decode(s, after, joinedCp);
293 cur = afterJoined;
294 continue;
295 }
296 break;
297 }
298 return cur;
299 }

307 inline size_t prevCluster(std::string_view s, size_t pos) noexcept
308 {
309 if (pos == 0)
310 return 0;
311 if (pos > s.size())
312 pos = s.size();
313
314 // If pos was mid-codepoint, snapping down already reaches a valid start < pos.
315 if (pos < s.size() && isContinuation(static_cast<unsigned char>(s[pos])))
316 return snapFloor(s, pos);
317
318 size_t cur = prevCodepoint(s, pos);
319 uint32_t cp = 0;
320 decode(s, cur, cp);
321
322 // CRLF as a single cluster.
323 if (cp == '\n' && cur > 0 && s[cur - 1] == '\r')
324 return cur - 1;
325
326 // Step back over any trailing extenders or ZWJ sequences.
327 while (cur > 0)
328 {
329 if (isClusterExtend(cp))
330 {
331 cur = prevCodepoint(s, cur);
332 decode(s, cur, cp);
333 continue;
334 }
335
336 // If immediately preceded by a ZWJ, step back over ZWJ + previous base.
337 size_t before = prevCodepoint(s, cur);
338 uint32_t beforeCp = 0;
339 decode(s, before, beforeCp);
340 if (beforeCp == 0x200Du && before > 0)
341 {
342 cur = prevCodepoint(s, before);
343 decode(s, cur, cp);
344 continue;
345 }
346 break;
347 }
348
349 // Regional Indicator pair: pair from the start of a consecutive RI run.
350 if (isRegionalIndicator(cp) && cur > 0)
351 {
352 size_t runCount = 0;
353 size_t scan = cur;
354 while (scan > 0)
355 {
356 size_t p = prevCodepoint(s, scan);
357 uint32_t pcp = 0;
358 decode(s, p, pcp);
359 if (!isRegionalIndicator(pcp))
360 break;
361 ++runCount;
362 scan = p;
363 }
364 if ((runCount % 2u) == 1u)
365 cur = prevCodepoint(s, cur);
366 }
367
368 return cur;
369 }

413 inline constexpr bool isWordChar(uint32_t cp) noexcept
414 {
415 if ((cp >= 'a' && cp <= 'z') || (cp >= 'A' && cp <= 'Z') ||
416 (cp >= '0' && cp <= '9') || cp == '_')
417 return true;
418 if (cp >= 0x00C0u && cp <= 0x00FFu)
419 return cp != 0x00D7u && cp != 0x00F7u; // exclude × and ÷
420 if (cp >= 0x0100u && cp <= 0x024Fu)
421 return true; // Latin Extended-A/B
422 if (isClusterExtend(cp))
423 return true;
424 if ((cp >= 0x0370u && cp <= 0x052Fu) || // Greek, Cyrillic
425 (cp >= 0x3040u && cp <= 0x30FFu) || // Hiragana, Katakana
426 (cp >= 0x4E00u && cp <= 0x9FFFu) || // CJK Unified Ideographs
427 (cp >= 0xAC00u && cp <= 0xD7AFu)) // Hangul Syllables
428 return true;
429 return false;
430 }
